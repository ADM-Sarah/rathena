-	script	Freebies_Package	-1,{

OnPCLoginEvent:

	// 1 ACCOUNT = 1 TIME
	if (#FreebiesPackage1 > 0) end;

	// 1 IP = 1 TIME (optional)
	.@ip$ = getcharip();
	if (getd("$FreebiesIP_" + .@ip$) > 0) end;

	// Optional: only for brand new character
	if (BaseLevel > 1 || JobLevel > 1) end;

	// ===== GIVE ITEMS =====
	getitembound 4147,1,Bound_Account;
	getitembound 33027,1,Bound_Account;
	getitembound 33028,1,Bound_Account;
	getitembound 33029,1,Bound_Account;
	getitembound 33031,1,Bound_Account;
	getitem 1208,1;
	getitem 33017,10;      // credit
	getitem 33022,10;     // ygg seed box
	getitem 33021,10;	//ygg bery box
	getitembound 51012,1,Bound_Account; // costume
	getitem 55677,1; // Weapon Pack

	set Zeny, Zeny + 10000000;
	dispbottom "You received 10,000,000 zeny.";

	// ===== MARK CLAIMED (IMPORTANT) =====
	set #FreebiesPackage1, 1;              // account flag
	setd "$FreebiesIP_" + .@ip$, 1;        // ip flag

	dispbottom "Welcome and Enjoy playing Nostalgic Ragnarok Online.";
	end;
}
