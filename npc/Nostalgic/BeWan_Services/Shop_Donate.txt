prt_in,59,143,5	script	Donate Manager	10132,{

OnAtDonate:
switch(select("~ Sale", "~ Headgears", "~ Armor", "~ Weapon", "~ Limited Costumes", "~ Rentals", "~ Usables", "~ Other", "~ ^FF0000Convert Points / Coin^000000")){
	case 1:
		if(.Sale == 1){
			callshop "cash_sale",1;
			end;
		}else{
			mes "[Donate Manager]";
			mes "Sorry, this is not available at the moment.";
			close;
		}
	break;

	case 2:
		callshop "cash_headgear",1;
		end;
	break;

	case 3:
		callshop "cash_armor",1;
		end;
	break;

	case 4:
		callshop "cash_weapon",1;
		end;
	break;

	case 5:
		callshop "cash_costume",1;
		end;
	break;

	case 6:
		if(.Rental == 1){
			callshop "cash_rental",1;
			end;
		}else{
			mes "[Donate Manager]";
			mes "Sorry, this is not available at the moment.";
			close;
		}

	break;

	case 7:
		callshop "cash_usable",1;
		end;
	break;

	case 8:
		callshop "cash_other",1;
		end;
	break;

	case 9:
		switch(select(""+.CpAmount+" Cashpoint = 1x "+getitemname(.ItemID)+"", "1x "+getitemname(.ItemID)+" = "+.CpAmount+" Cashpoint")){
			case 1:
				mes "[Donate Manager]";
				mes "Your current cashpoints is: "+#CASHPOINTS+".";
    				mes "How many cashpoints do you want to exchange?";
    				next;
    				input .@amount;

				if (.@amount <= 0) {
    					mes "[Donate Manager]";
    					mes "Umm, please enter your amount!";
    					close;
				}else if(#CASHPOINTS < .@amount){
					mes "[Donate Manager]";
					mes "You dont' have enough cash points.";
					mes "You input amount is: "+.@amount+" and your cash points is: "+#CASHPOINTS+".";
					close;
				}else{
					mes "[Donate Manager]";
					mes "Here you go!";
					set .TotalCP,.@amount * .CpAmount;
					set #CASHPOINTS,#CASHPOINTS - .TotalCP;
					getitem .ItemID,.TotalCP;
					dispbottom "You got "+.TotalCP+" "+getitemname(.ItemID)+"";
					dispbottom "Your total cash points left is: "+#CASHPOINTS+"";
					close;
				}

			case 2:
				mes "[Donate Manager]";
    				mes "How many "+getitemname(.ItemID)+" do you have?";
    				next;
    				input .@amount;

				if (.@amount <= 0) {
    					mes "[Donate Manager]";
    					mes "Umm, please enter your amount!";
    					close;
				}else if(countitem(.ItemID) < .@amount){
					mes "[Donate Manager]";
					mes "You dont' have enough "+getitemname(.ItemID)+"!";
					close;
				}else{
					if(countitem(.ItemID) < .@amount){
						mes "[Donate Manager]";
						mes "You dont' have enough "+getitemname(.ItemID)+"!";
						close;
					}else{
						mes "[Donate Manager]";
						mes "Here you go!";
						set .TotalCP,.@amount * .CpAmount;
						set #CASHPOINTS,#CASHPOINTS + .TotalCP;
						delitem .ItemID,.@amount;
						dispbottom "You got "+.TotalCP+" Cash Points!";
						dispbottom "Your total cash points is: "+#CASHPOINTS+"";
						close;
					}
				}
		}

}
end;

OnInit:
	set .Sale, 0; //0 = disable and 1 = enable
	set .Rental, 1; //0 = disable and 1 = enable
	set .ItemID, 33048;
	set .CpAmount, 1; //Cashpoints amount
	bindatcmd("donate", strnpcinfo(3)+"::OnAtDonate");
	//waitingroom "Donate Manager",0;
	end;
}
