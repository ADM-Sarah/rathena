// Shop NPCs -- supplying no argument displays entire menu.
//============================================================
valmorra,147,222,4	script	Donation Set Manager#1	11007,{

OnClick:
	goto OnMenu;

OnInit:
	freeloop(1);

	.Announce = 0;
	.ShowSlot = 1;
	.ShowID = 0;
	.ShowZeny = 0;
	.MaxStack = 1;
	.ignore_secure_npctimeout = 1;

	.CommissionCoinID = 33048;

	.BonusPotionBoxID = 51001;
	.BonusBBBoxID = 51002;
	.BonusBubbleGumBoxID = 12902;
	.BonusVip3DayID = 33035;
	.BonusVip7DayID = 33036;
	.BonusVip30DayID = 33037;
	.BonusEnrichOriID = 7620;
	.BonusEnrichEluID = 7619;
	.BonusExclusiveCostumeID = 55676;
	.BonusNostaCoinID = 33048;

	setarray .MilestoneCost[0], 100,500,1000,2000,3000,3500;
	setarray .MilestoneMask[0], 1,2,4,8,16,32;

	setarray .MilestonePotion[0], 25,50,100,150,250,300;
	setarray .MilestoneNostaCoin[0], 15,50,100,150,200,250;
	setarray .MilestoneVip3D[0], 3,4,0,0,0,0;
	setarray .MilestoneVip7D[0], 0,0,4,0,0,0;
	setarray .MilestoneVip30D[0], 0,0,0,3,4,5;
	setarray .MilestoneBB[0], 2,3,5,7,10,15;
	setarray .MilestoneBubble[0], 2,3,5,7,10,15;
	setarray .MilestoneOri[0], 0,10,20,25,30,35;
	setarray .MilestoneElu[0], 0,10,20,25,30,35;
	setarray .MilestoneCostume[0], 0,0,0,0,0,1;

	.DialogW = 430;
	.DialogH = 430;

	setarray .Points$[0],"#HEADGEARPOINTS","Headgear Points";

	setarray .Shops$[1],
	"~ ^E74C3CRed Regin^000000 - ^E74C3C[Damage Set]^000000",
	"~ ^555555Black Regin^000000 - ^555555[Tanky Set]^000000",
	"~ ^D4AF37Gold Regin^000000 - ^D4AF37[Magic Set]^000000",
	"~ ^2E8B57Green Regin^000000 - ^2E8B57[Range Set]^000000";

	callsub S_Add,1,50024,1,0,0,33048,100;
	callsub S_Add,1,50192,1,0,0,33048,100;
	callsub S_Add,1,50348,1,0,0,33048,100;

	callsub S_Add,2,50021,1,0,0,33048,100;
	callsub S_Add,2,50187,1,0,0,33048,100;
	callsub S_Add,2,50343,1,0,0,33048,100;

	callsub S_Add,3,50026,1,0,0,33048,100;
	callsub S_Add,3,50189,1,0,0,33048,100;
	callsub S_Add,3,50346,1,0,0,33048,100;

	callsub S_Add,4,50023,1,0,0,33048,100;
	callsub S_Add,4,50190,1,0,0,33048,100;
	callsub S_Add,4,50347,1,0,0,33048,100;

	freeloop(0);

	.menu$ = "";
	for(.@i = 1; .@i <= getarraysize(.Shops$); .@i++) {
		.menu$ += .Shops$[.@i] + ":";
		npcshopdelitem "donationset11"+.@i,909;
	}
	end;

OnMenu:
	if (.ignore_secure_npctimeout)
		ignoretimeout 1;

	setdialogsize(.DialogW,.DialogH);
	setdialogalign(DIALOG_ALIGN_CENTER);

	.@aid = getcharid(3);
	.@total_spent = 0;
	query_sql("SELECT `total_spent` FROM `streamer_bonus_spend` WHERE `account_id` = " + .@aid + " LIMIT 1", .@total_spent);

	.@next_ms = callsub(S_NextMilestone,.@total_spent);
	.@need_ms = .@next_ms - .@total_spent;

	mes "<b>^000000[ ^FF7F00Donation Set Manager^000000 ]</b>";
	mes callsub(S_Line);
	mes "^000000Welcome to the ^FF7F00Donation Set Exchange^000000.";
	mes "^000000Each item requires ^1E90FF100 Nostalgic Coin^000000.";
	mes callsub(S_Line);
	mes "^000000Your Total Spend:^000000 ^00AA00" + .@total_spent + " Coin^000000";

	if (.@next_ms > 0)
		mes "^000000Next Milestone:^000000 ^FF9900" + .@next_ms + " Coin^000000 ^777777(" + .@need_ms + " more)^000000";
	else
		mes "^000000Next Milestone:^000000 ^00AA00All milestone rewards completed.^000000";

	mes callsub(S_Line);
	mes "^FF0000Reminder:^000000 Milestone rewards are counted from your total spending across supported Donation NPCs.";
	mes "^00AA00Milestone Bonus:^000000 Nostalgic Potion Box and Nostalgic Coin are included in milestone rewards.";
	mes "^777777Streamer Code system is currently disabled.^000000";
	mes callsub(S_Line);
	next;

	.@size = getarraysize(@i);
	if (!.@size) @shop_index = select(.menu$);
	else if (.@size == 1) @shop_index = @i[0];
	else {
		.@menu$ = "";
		for(.@j = 0; .@j < .@size; .@j++)
			.@menu$ += .Shops$[@i[.@j]] + ":";
		@shop_index = @i[select(.@menu$)-1];
	}
	deletearray @i[0],getarraysize(@i);

	if (.Shops$[@shop_index] == "") {
		message strcharinfo(0),"An error has occurred.";
		end;
	}

	dispbottom "Select one item at a time.";

	close2;
	callshop "donationset11"+@shop_index,1;
	npcshopattach "donationset11"+@shop_index;
	end;

OnBuyItem:
	if (.ignore_secure_npctimeout)
		ignoretimeout 1;

	setdialogsize(.DialogW,.DialogH);
	setdialogalign(DIALOG_ALIGN_CENTER);

	setarray .@q[0],@bought_nameid[0],((@bought_quantity[0] > .MaxStack) ? .MaxStack : @bought_quantity[0]);
	copyarray .@q[3],getd(".q_"+@shop_index+"_"+.@q[0]+"[0]"),getarraysize(getd(".q_"+@shop_index+"_"+.@q[0]));
	.@q[2] = .@q[1] * .@q[3];

	if (!.@q[2] || .@q[2] > 30000) {
		message strcharinfo(0),"You can't purchase that many "+getitemname(.@q[0])+".";
		end;
	}

	mes "<b>^000000[ ^FF7F00Donation Set Manager^000000 ]</b>";
	mes callsub(S_Line);
	mes "^000000Shop:^000000 ^FF7F00" + .Shops$[@shop_index] + "^000000";
	mes callsub(S_Line);

	mes "<b>^000000Reward:^000000</b> ^0055FF"
		+ ((.@q[2] > 1) ? .@q[2]+"x " : "")
		+ "^i[" + .@q[0] + "] "
		+ callsub(S_Slot,.@q[0])
		+ "^000000";

	mes "<b>^000000Requirements:^000000</b>";
	disable_items;

	set @qe[0],0;

	if (.@q[4])
		mes "^3CB371»^000000 " + callsub(S_Chk,Zeny,.@q[4]*.@q[1]) + F_InsertComma((.@q[4]*.@q[1])) + " Zeny^000000";

	if (.@q[5])
		mes "^3CB371»^000000 " + callsub(S_Chk,getd(.Points$[0]),.@q[5]*.@q[1]) + (.@q[5]*.@q[1])+" "+.Points$[1]+" ^777777("+getd(.Points$[0])+"/"+(.@q[5]*.@q[1])+")^000000";

	if (.@q[6]) {
		for(.@i = 6; .@i < getarraysize(.@q); .@i += 2) {
			mes "^3CB371»^000000 "
				+ callsub(S_Chk,countitem(.@q[.@i]),.@q[.@i+1]*.@q[1])
				+ ((.ShowID)?"{"+.@q[.@i]+"} ":"")
				+ "^i[" + .@q[.@i] + "] "
				+ "<ITEM>" + callsub(S_Slot,.@q[.@i]) + "<INFO>" + .@q[.@i] + "</INFO></ITEM> "
				+ "^777777(" + countitem(.@q[.@i]) + "/" + (.@q[.@i+1]*.@q[1]) + ")^000000";
		}
	}

	.@preview_spent = 0;
	if (.@q[6]) {
		for(.@i = 6; .@i < getarraysize(.@q); .@i += 2) {
			if (.@q[.@i] == .CommissionCoinID)
				.@preview_spent += .@q[.@i+1] * .@q[1];
		}
	}

	.@aid = getcharid(3);
	.@old_total_preview = 0;
	query_sql("SELECT `total_spent` FROM `streamer_bonus_spend` WHERE `account_id` = " + .@aid + " LIMIT 1", .@old_total_preview);

	.@after_total_preview = .@old_total_preview + .@preview_spent;
	.@next_ms_after = callsub(S_NextMilestone,.@after_total_preview);
	.@need_ms_after = .@next_ms_after - .@after_total_preview;

	mes callsub(S_Line);
	mes "^000000Spend This Purchase:^000000 ^00AA00" + .@preview_spent + " Coin^000000";
	mes "^000000Total After Purchase:^000000 ^00AA00" + .@after_total_preview + " Coin^000000";

	if (.@next_ms_after > 0)
		mes "^000000Next Milestone:^000000 ^FF9900" + .@next_ms_after + " Coin^000000 ^777777(" + .@need_ms_after + " more)^000000";
	else
		mes "^000000Next Milestone:^000000 ^00AA00All milestone rewards completed.^000000";

	setdialogsize(.DialogW,.DialogH);
	setdialogalign(DIALOG_ALIGN_CENTER);
	next;

	.@bonus_box = 0;
	.@bonus_nosta = 0;
	.@bonus_vip3 = 0;
	.@bonus_vip7 = 0;
	.@bonus_vip30 = 0;
	.@bonus_bb = 0;
	.@bonus_bubble = 0;
	.@bonus_ori = 0;
	.@bonus_elu = 0;
	.@bonus_costume = 0;
	.@spent_coin = 0;

	setarray @qe[1], getiteminfo(.@q[0],5), getiteminfo(.@q[0],11);


	if (@qe[2] > 0 && ((@qe[1] & 1) || (@qe[1] & 256) || (@qe[1] & 512) || (@qe[1] & 1024) || (@qe[1] & 2048) || (@qe[1] & 4096) || (@qe[1] & 4) || (@qe[1] & 8192)))
		.@preview = 1;

	addtimer 1000, strnpcinfo(3)+"::OnEnd";

	while(1) {
		switch(select(
			"^3CB371» Purchase ^0055FF"+getitemname(.@q[0])+"^000000"
			+ ":"
			+ ((.@preview && !@qe[7]) ? "^1E90FF» Preview Item^000000:" : "")
			+ "^FF0000» Cancel^000000"
		)) {
		case 1:
			if (@qe[0]) {
				mes "^4B0082------------------------------^000000";
				mes "^4B0082  Donation Set Manager^000000";
				mes "^4B0082------------------------------^000000";
				mes "^FF0000You're missing one or more quest requirements.^000000";
				close;
			}

			if (!checkweight(.@q[0],.@q[2])) {
				mes "^4B0082------------------------------^000000";
				mes "^4B0082  Donation Set Manager^000000";
				mes "^4B0082------------------------------^000000";
				mes "^FF0000You need more weight capacity to complete this trade.^000000";
				close;
			}

			if (.@q[4]) Zeny -= (.@q[4]*.@q[1]);
			if (.@q[5]) setd .Points$[0], getd(.Points$[0])-(.@q[5]*.@q[1]);

			if (.@q[6]) {
				for(.@i = 6; .@i < getarraysize(.@q); .@i += 2) {
					if (.@q[.@i] == .CommissionCoinID)
						.@spent_coin += .@q[.@i+1] * .@q[1];

					delitem .@q[.@i], .@q[.@i+1]*.@q[1];
				}
			}

			getitem .@q[0], .@q[2];

			if (.@spent_coin > 0) {
				.@aid = getcharid(3);
				.@cid = getcharid(0);
				.@name$ = escape_sql(strcharinfo(0));

				.@old_total = 0;
				.@claimed_100 = 0;
				.@claimed_1000 = 0;
				.@claimed_mask = 0;

				query_sql(
					"SELECT `total_spent`, `claimed_100_tiers`, `claimed_1000_tiers`, `claimed_milestone_mask` FROM `streamer_bonus_spend` WHERE `account_id` = " + .@aid + " LIMIT 1",
					.@old_total, .@claimed_100, .@claimed_1000, .@claimed_mask
				);

				.@new_total = .@old_total + .@spent_coin;

				for (.@i = 0; .@i < getarraysize(.MilestoneCost); .@i++) {
					if (.@new_total >= .MilestoneCost[.@i] && !(.@claimed_mask & .MilestoneMask[.@i])) {

						if (.MilestonePotion[.@i] > 0) {
							getitem .BonusPotionBoxID, .MilestonePotion[.@i];
							.@bonus_box += .MilestonePotion[.@i];
						}

						if (.MilestoneNostaCoin[.@i] > 0) {
							getitem .BonusNostaCoinID, .MilestoneNostaCoin[.@i];
							.@bonus_nosta += .MilestoneNostaCoin[.@i];
						}

						if (.MilestoneVip3D[.@i] > 0) {
							getitem .BonusVip3DayID, .MilestoneVip3D[.@i];
							.@bonus_vip3 += .MilestoneVip3D[.@i];
						}

						if (.MilestoneVip7D[.@i] > 0) {
							getitem .BonusVip7DayID, .MilestoneVip7D[.@i];
							.@bonus_vip7 += .MilestoneVip7D[.@i];
						}

						if (.MilestoneVip30D[.@i] > 0) {
							getitem .BonusVip30DayID, .MilestoneVip30D[.@i];
							.@bonus_vip30 += .MilestoneVip30D[.@i];
						}

						if (.MilestoneBB[.@i] > 0) {
							getitem .BonusBBBoxID, .MilestoneBB[.@i];
							.@bonus_bb += .MilestoneBB[.@i];
						}

						if (.MilestoneBubble[.@i] > 0) {
							getitem .BonusBubbleGumBoxID, .MilestoneBubble[.@i];
							.@bonus_bubble += .MilestoneBubble[.@i];
						}

						if (.MilestoneOri[.@i] > 0) {
							getitem .BonusEnrichOriID, .MilestoneOri[.@i];
							.@bonus_ori += .MilestoneOri[.@i];
						}

						if (.MilestoneElu[.@i] > 0) {
							getitem .BonusEnrichEluID, .MilestoneElu[.@i];
							.@bonus_elu += .MilestoneElu[.@i];
						}

						if (.MilestoneCostume[.@i] > 0) {
							getitem .BonusExclusiveCostumeID, .MilestoneCostume[.@i];
							.@bonus_costume += .MilestoneCostume[.@i];
						}

						.@claimed_mask |= .MilestoneMask[.@i];
					}
				}

				query_sql(
					"INSERT INTO `streamer_bonus_spend` (`account_id`,`char_id`,`name`,`total_spent`,`claimed_100_tiers`,`claimed_1000_tiers`,`claimed_milestone_mask`,`updated_at`) VALUES (" +
					.@aid + "," + .@cid + ",'" + .@name$ + "'," + .@new_total + "," + .@claimed_100 + "," + .@claimed_1000 + "," + .@claimed_mask + ",NOW()) " +
					"ON DUPLICATE KEY UPDATE " +
					"`char_id` = VALUES(`char_id`), " +
					"`name` = VALUES(`name`), " +
					"`total_spent` = " + .@new_total + ", " +
					"`claimed_milestone_mask` = " + .@claimed_mask + ", " +
					"`updated_at` = NOW()"
				);

			}

			if (.Announce)
				announce strcharinfo(0)+" has created "+((.@q[2] > 1)? .@q[2]+"x "+getitemname(.@q[0]) : callfunc("F_InsertArticle",getitemname(.@q[0])))+"!",0;

			specialeffect2 EF_FLOWERLEAF;

			mes "^4B0082------------------------------^000000";
			mes "^4B0082  Donation Set Manager^000000";
			mes "^4B0082------------------------------^000000";
			mes "^1E90FFTrade completed successfully!^000000";
			mes callsub(S_Line);

			mes "^000000You received:^000000 ^0055FF"
				+ ((.@q[2] > 1) ? .@q[2]+"x " : "")
				+ "^i[" + .@q[0] + "] "
				+ callsub(S_Slot,.@q[0])
				+ "^000000";

			if (.@spent_coin > 0) {
				mes "^000000Total Spend Updated:^000000 ^00AA00" + .@new_total + " Coin^000000";

				.@next_ms_final = callsub(S_NextMilestone,.@new_total);
				.@need_ms_final = .@next_ms_final - .@new_total;

				if (.@next_ms_final > 0)
					mes "^000000Next Milestone:^000000 ^FF9900" + .@next_ms_final + " Coin^000000 ^777777(" + .@need_ms_final + " more)^000000";
				else
					mes "^000000Next Milestone:^000000 ^00AA00All milestone rewards completed.^000000";
			}

			if (.@bonus_box > 0 || .@bonus_nosta > 0) {
				mes "^000000Milestone Bonus:^000000";

				if (.@bonus_box > 0)
					mes "^3CB371»^000000 ^i[" + .BonusPotionBoxID + "] " + getitemname(.BonusPotionBoxID) + " x" + .@bonus_box;

				if (.@bonus_nosta > 0)
					mes "^3CB371»^000000 ^i[" + .BonusNostaCoinID + "] " + getitemname(.BonusNostaCoinID) + " x" + .@bonus_nosta;
			}


			if (.@bonus_vip3 > 0 || .@bonus_vip7 > 0 || .@bonus_vip30 > 0 || .@bonus_bb > 0 || .@bonus_bubble > 0 || .@bonus_ori > 0 || .@bonus_elu > 0 || .@bonus_costume > 0) {
				if (.@bonus_box <= 0 && .@bonus_nosta <= 0)
					mes "^000000Milestone Bonus:^000000";

				if (.@bonus_vip3 > 0)
					mes "^3CB371»^000000 ^i[" + .BonusVip3DayID + "] " + getitemname(.BonusVip3DayID) + " x" + .@bonus_vip3;

				if (.@bonus_vip7 > 0)
					mes "^3CB371»^000000 ^i[" + .BonusVip7DayID + "] " + getitemname(.BonusVip7DayID) + " x" + .@bonus_vip7;

				if (.@bonus_vip30 > 0)
					mes "^3CB371»^000000 ^i[" + .BonusVip30DayID + "] " + getitemname(.BonusVip30DayID) + " x" + .@bonus_vip30;

				if (.@bonus_bb > 0)
					mes "^3CB371»^000000 ^i[" + .BonusBBBoxID + "] " + getitemname(.BonusBBBoxID) + " x" + .@bonus_bb;

				if (.@bonus_bubble > 0)
					mes "^3CB371»^000000 ^i[" + .BonusBubbleGumBoxID + "] " + getitemname(.BonusBubbleGumBoxID) + " x" + .@bonus_bubble;

				if (.@bonus_ori > 0)
					mes "^3CB371»^000000 ^i[" + .BonusEnrichOriID + "] " + getitemname(.BonusEnrichOriID) + " x" + .@bonus_ori;

				if (.@bonus_elu > 0)
					mes "^3CB371»^000000 ^i[" + .BonusEnrichEluID + "] " + getitemname(.BonusEnrichEluID) + " x" + .@bonus_elu;

				if (.@bonus_costume > 0)
					mes "^3CB371»^000000 ^i[" + .BonusExclusiveCostumeID + "] " + getitemname(.BonusExclusiveCostumeID) + " x" + .@bonus_costume;
			}

			close;

		case 2:
			if (.@preview && !@qe[7]) {
				setarray @qe[3], getlook(LOOK_HEAD_BOTTOM), getlook(LOOK_HEAD_TOP), getlook(LOOK_HEAD_MID), getlook(LOOK_ROBE), 1;

				if ((@qe[1] & 1) || (@qe[1] & 4096))
					changelook LOOK_HEAD_BOTTOM, @qe[2];
				else if ((@qe[1] & 256) || (@qe[1] & 1024))
					changelook LOOK_HEAD_TOP, @qe[2];
				else if ((@qe[1] & 512) || (@qe[1] & 2048))
					changelook LOOK_HEAD_MID, @qe[2];
				else if ((@qe[1] & 4) || (@qe[1] & 8192))
					changelook LOOK_ROBE, @qe[2];

				break;
			}
			close;

		case 3:
			close;
		}
	}

OnEnd:
	if (.ignore_secure_npctimeout)
		ignoretimeout 0;

	if (@qe[7]) {
		changelook LOOK_HEAD_BOTTOM, @qe[3];
		changelook LOOK_HEAD_TOP, @qe[4];
		changelook LOOK_HEAD_MID, @qe[5];
		changelook LOOK_ROBE, @qe[6];
	}

	deletearray @qe[0],8;
	end;

S_NextMilestone:
	.@spent = getarg(0);

	for (.@i = 0; .@i < getarraysize(.MilestoneCost); .@i++) {
		if (.@spent < .MilestoneCost[.@i])
			return .MilestoneCost[.@i];
	}
	return 0;

S_Add:
	if (getitemname(getarg(1)) == "null") {
		debugmes "Quest reward #"+getarg(1)+" invalid (skipped).";
		return;
	}

	setarray .@j[0],getarg(2),getarg(3),getarg(4);

	for(.@i = 5; .@i < getargcount(); .@i += 2) {
		if (getitemname(getarg(.@i)) == "null") {
			debugmes "Quest requirement #"+getarg(.@i)+" invalid (skipped).";
			return;
		}
		setarray .@j[.@i-2], getarg(.@i), getarg(.@i+1);
	}

	copyarray getd(".q_"+getarg(0)+"_"+getarg(1)+"[0]"), .@j[0], getarraysize(.@j);
	npcshopadditem "donationset11"+getarg(0), getarg(1), ((.ShowZeny)?getarg(3):0);
	return;

S_Chk:
	if (getarg(0) < getarg(1)) {
		@qe[0] = 1;
		return "^FF0000";
	}
	return "^00BFFF";

S_Slot:
	.@s$ = getitemname(getarg(0));
	switch(.ShowSlot) {
		case 1:
			if (!getitemslots(getarg(0))) return .@s$;
		case 2:
			if (getiteminfo(getarg(0),2) == 4 || getiteminfo(getarg(0),2) == 5)
				return .@s$+" ["+getitemslots(getarg(0))+"]";
		default:
			return .@s$;
	}

S_Line:
	return "^777777--------------------------------------------------^000000";
}

-	shop	donationset111	-1,909:-1

-	shop	donationset114	-1,909:-1
