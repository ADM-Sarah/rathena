	query_sql "CREATE TABLE IF NOT EXISTS `goldroom_device_ip_limit` (`account_id` INT(11) NOT NULL, `char_id` INT(11) NOT NULL, `ip_address` VARCHAR(64) NOT NULL, `unique_id` VARCHAR(64) NOT NULL, `updated` INT(11) NOT NULL DEFAULT '0', PRIMARY KEY (`char_id`), KEY `ip_address` (`ip_address`), KEY `unique_id` (`unique_id`)) ENGINE=MyISAM";

	setmapflag .map$,mf_nosave;
	setmapflag .map$,mf_nomemo;
	setmapflag .map$,mf_nobranch;
	setmapflag .map$,mf_noexp;
	setmapflag .map$,mf_nochat;
	setmapflag .map$,mf_novending;
	setmapflag .map$,mf_nowarpto;
	setmapflag .map$,mf_noreturn;
	setmapflag .map$,mf_nopenalty;
	setmapflag .map$,mf_noskill;
	setmapflag .map$,mf_loadevent;

	bindatcmd "gron",   strnpcinfo(0)+"::OnGoldCommand",     1,99;
	bindatcmd "groff",  strnpcinfo(0)+"::OnGoldOffCommand",  1,99;
	bindatcmd "grtime", strnpcinfo(0)+"::OnGoldTimeCommand", 1,99;

	donpcevent strnpcinfo(0)+"::OnRecoverBoostState";
	end;

OnPCLoadMapEvent:
	if (strcharinfo(3) != .map$) end;

	// GM 99 bypass
	if (getgroupid() >= 99) end;

	// If player somehow enters through warp/command, re-check inside map.
	callfunc "F_GoldRoom_DeviceIPLimit", 1;
	end;

OnCheckBoostExpire:
	if (!$GOLDROOM_ACTIVE) end;

	.@now = gettimetick(2);
	if ($GOLDROOM_END <= 0 || .@now >= $GOLDROOM_END) {
		$GOLDROOM_ACTIVE = 0;
		$GOLDROOM_START = 0;
		$GOLDROOM_END = 0;
		announce "Gold Boost has ended. Gold Room is back to normal.", bc_all, 0xFFA500;
	}
	end;

OnRecoverBoostState:
	if (!$GOLDROOM_ACTIVE) end;

	.@now = gettimetick(2);
	if ($GOLDROOM_START <= 0 || $GOLDROOM_END <= 0 || .@now >= $GOLDROOM_END) {
		$GOLDROOM_ACTIVE = 0;
		$GOLDROOM_START = 0;
		$GOLDROOM_END = 0;
		end;
	}

	.@remain = $GOLDROOM_END - .@now;
	announce "Gold Boost is still active after server restart/reload! Remaining time: "
		+ sprintf("%02d hour(s) %02d minute(s) %02d second(s)", .@remain / 3600, (.@remain % 3600) / 60, .@remain % 60), bc_all, 0xFFA500;
	end;

OnClock0100:
OnClock0300:
OnClock1300:
OnClock1900:
OnClock2300:
	donpcevent strnpcinfo(0)+"::OnCheckBoostExpire";
	if ($GOLDROOM_ACTIVE) end;
	donpcevent strnpcinfo(0)+"::OnAutoBoostStart";
	end;

OnAutoBoostStart:
	if ($GOLDROOM_ACTIVE) end;

	$GOLDROOM_ACTIVE = 1;
	$GOLDROOM_START = gettimetick(2);
	$GOLDROOM_END = $GOLDROOM_START + .boost_duration;

	announce "Gold Boost has started for 1 hour! Every kill now gives random bonus Gold x2~x5!", bc_all, 0xFFA500;
	end;

OnGoldKill:
	donpcevent strnpcinfo(0)+"::OnCheckBoostExpire";

	if (strcharinfo(3) != .map$) end;
	if (!$GOLDROOM_ACTIVE) end;

	.@amount = rand(.boost_min, .boost_max);

	if (!checkweight(.gold_item, .@amount)) {
		dispbottom "[Gold Room] Your inventory is full or overweight. Gold Boost reward was not added.";
		dispbottom "[Gold Room] Please clear your inventory or reduce weight first.";
		end;
	}

	getitem .gold_item, .@amount;
	dispbottom "Gold Boost reward: Gold x" + .@amount;
	end;

OnGoldCommand:
	donpcevent strnpcinfo(0)+"::OnCheckBoostExpire";

	if ($GOLDROOM_ACTIVE) {
		dispbottom "Gold Room boost is already active.";
		end;
	}

	$GOLDROOM_ACTIVE = 1;
	$GOLDROOM_START = gettimetick(2);
	$GOLDROOM_END = $GOLDROOM_START + .boost_duration;

	announce "GM " + strcharinfo(0) + " has activated Gold Boost for 1 hour! Every kill now gives random bonus Gold x2~x5!", bc_all, 0xFFA500;
	dispbottom "Gold Room boost started manually for 1 hour.";
	end;

OnGoldOffCommand:
	donpcevent strnpcinfo(0)+"::OnCheckBoostExpire";

	if (!$GOLDROOM_ACTIVE) {
		dispbottom "Gold Room boost is not active.";
		end;
	}

	$GOLDROOM_ACTIVE = 0;
	$GOLDROOM_START = 0;
	$GOLDROOM_END = 0;

	announce "GM " + strcharinfo(0) + " has ended Gold Boost manually. Gold Room is back to normal.", bc_all, 0xFFA500;
	end;

OnGoldTimeCommand:
	donpcevent strnpcinfo(0)+"::OnCheckBoostExpire";

	if ($GOLDROOM_ACTIVE) {
		.@remain = $GOLDROOM_END - gettimetick(2);
		if (.@remain < 0) .@remain = 0;

		dispbottom "Gold Boost status: ACTIVE";
		dispbottom "Time remaining: " + sprintf("%02d hour(s) %02d minute(s) %02d second(s)", .@remain / 3600, (.@remain % 3600) / 60, .@remain % 60);
		dispbottom "Boost reward: random Gold x2~x5 per kill";
		dispbottom "Normal reward: monster DB drop only";
		dispbottom "Fixed boost times: 01:00, 03:00, 13:00, 19:00, 23:00";
		end;
	}

	dispbottom "Gold Boost status: INACTIVE";
	dispbottom "Normal reward: monster DB drop only";
	dispbottom "Boost reward: random Gold x2~x5 per kill";
	dispbottom "Fixed boost times: 01:00, 03:00, 13:00, 19:00, 23:00";
	end;
}

//==================================================
// Gold Room IP + Device Limit - SQL Safe Version
// - Max 5 IDs/characters per same IP Address
// - Max 5 IDs/characters per same Gepard Unique ID / Device
// - Helps reduce VPN bypass because device Unique ID is also checked
// - No getunits command used, suitable for older rAthena
// - Mode 0: check before entering Gold Room
// - Mode 1: check after loading Gold Room map, reject if over limit
//==================================================
function	script	F_GoldRoom_DeviceIPLimit	{
	.@mode = getarg(0,0); // 0 = before warp, 1 = inside map load
	.@map$ = "ordeal_3-1";
	.@limit = 5;

	// GM 99 bypass
	if (getgroupid() >= 99)
		return;

	.@my_aid = getcharid(3);
	.@my_cid = getcharid(0);
	.@my_ip$ = getcharip();
	.@my_uid$ = get_unique_id();

	if (.@my_ip$ == "") {
		if (.@mode == 0) {
			mes "[Gold Room]";
			mes "^FF0000IP Address not detected.^000000";
			mes "Please relog and try again.";
			close;
		}
		dispbottom "[Gold Room] IP Address not detected. Please relog and try again.";
		warp "SavePoint",0,0;
		end;
	}

	if (.@my_uid$ == "") {
		if (.@mode == 0) {
			mes "[Gold Room]";
			mes "^FF0000Device ID not detected.^000000";
			mes "Please relog and try again.";
			close;
		}
		dispbottom "[Gold Room] Device ID not detected. Please relog and try again.";
		warp "SavePoint",0,0;
		end;
	}

	// Clean old records that are no longer online or no longer inside Gold Room.
	query_sql "DELETE gr FROM `goldroom_device_ip_limit` gr LEFT JOIN `char` c ON gr.`char_id` = c.`char_id` WHERE c.`char_id` IS NULL OR c.`online` = '0' OR c.`last_map` <> '"+.@map$+"'";

	// Count current same IP inside Gold Room.
	query_sql "SELECT COUNT(*) FROM `goldroom_device_ip_limit` gr INNER JOIN `char` c ON gr.`char_id` = c.`char_id` WHERE gr.`ip_address` = '"+escape_sql(.@my_ip$)+"' AND c.`online` = '1' AND c.`last_map` = '"+.@map$+"'", .@ip_count;

	// Count current same Device / Gepard Unique ID inside Gold Room.
	query_sql "SELECT COUNT(*) FROM `goldroom_device_ip_limit` gr INNER JOIN `char` c ON gr.`char_id` = c.`char_id` WHERE gr.`unique_id` = '"+escape_sql(.@my_uid$)+"' AND c.`online` = '1' AND c.`last_map` = '"+.@map$+"'", .@uid_count;

	// Before warp: block if same IP or same Device already has 5 inside.
	if (.@mode == 0 && (.@ip_count >= .@limit || .@uid_count >= .@limit)) {
		mes "[Gold Room]";
		mes "^FF0000Gold Room Security Limit^000000";
		mes "Max ^0000FF5 IDs^000000 only can enter this Gold Room.";
		mes "Please leave with one of your characters first before entering again.";
		close;
	}

	// Inside map load: register current character first.
	if (.@mode == 1) {
		query_sql "REPLACE INTO `goldroom_device_ip_limit` (`account_id`, `char_id`, `ip_address`, `unique_id`, `updated`) VALUES ('"+.@my_aid+"', '"+.@my_cid+"', '"+escape_sql(.@my_ip$)+"', '"+escape_sql(.@my_uid$)+"', '"+gettimetick(2)+"')";

		query_sql "DELETE gr FROM `goldroom_device_ip_limit` gr LEFT JOIN `char` c ON gr.`char_id` = c.`char_id` WHERE c.`char_id` IS NULL OR c.`online` = '0' OR c.`last_map` <> '"+.@map$+"'";

		query_sql "SELECT COUNT(*) FROM `goldroom_device_ip_limit` gr INNER JOIN `char` c ON gr.`char_id` = c.`char_id` WHERE gr.`ip_address` = '"+escape_sql(.@my_ip$)+"' AND c.`online` = '1' AND c.`last_map` = '"+.@map$+"'", .@ip_count;

		query_sql "SELECT COUNT(*) FROM `goldroom_device_ip_limit` gr INNER JOIN `char` c ON gr.`char_id` = c.`char_id` WHERE gr.`unique_id` = '"+escape_sql(.@my_uid$)+"' AND c.`online` = '1' AND c.`last_map` = '"+.@map$+"'", .@uid_count;

		if (.@ip_count > .@limit || .@uid_count > .@limit) {
			query_sql "DELETE FROM `goldroom_device_ip_limit` WHERE `char_id` = '"+.@my_cid+"'";
			dispbottom "[Gold Room] Max 5 IDs only can enter this Gold Room.";
			dispbottom "[Gold Room] You have been moved out because the limit has been reached.";
			warp "SavePoint",0,0;
			end;
		}
	}

	return;
}
