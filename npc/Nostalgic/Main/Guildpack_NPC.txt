valmorra,230,48,4	script	GuildPack NPC	10191,{

	.@unique_id$ = get_unique_id();
	
	if( !getcharid(2) ){
		mes "You Don't Have Guild.";
		close;
	}

	mes "[ Nostalgic Staff ]";
	mes "Good day!";
	mes "Do you want to claim Guild-Pack?!";
	mes " ";
	mes "Must have atleast 10 members to claim for Guild Package.";
	mes "Take note! NostalgicRO Staff are not responsible of the items lost ";
	mes "provided in the guild pack if you left the guild.";
	next;
	menu "Yes Please",yes,"Nope",no;

yes:
	query_sql "SELECT `guild_id`, COUNT(`online`) FROM `char` WHERE `guild_id` = '"+getcharid(2)+"' AND `online` = '1'", .@Gid, .@online;
	query_sql "SELECT `GID` FROM `data_gpack` WHERE `GID` = '"+getcharid(2)+"'", .@GuildID;
	getinventorylist;
	.@weight = 100 * Weight / MaxWeight;
	if ( .@online < 10 ) {
		mes "[ Nostalgic Staff ]";
		mes "Sorry not enough online members";
		mes "You must have 10 members online to claim Guild-Pack";
		close;
	}else if ( @inventorylist_count > 30 ) {
		mes "[ Nostalgic Staff ]";
		mes "You have 30 different items in your inventory.";
		close;
	}else if((.@weight) > 50){
		mes "[ Nostalgic Staff ]";
		mes "^FF0000You must atleast have 50 percent extra weight to receive guildpack.^000000";
		mes " ";
		mes "You are currently at " + .@weight + " percent of your maximum weight.";
		close;
	}else if(getd("$GpackID_" + .@unique_id$) > 0 || #GuildPackReward > 0){
		mes "[ Nostalgic Staff ]";
		mes "You have already claimed your GuildPack.";
		close;
	}else{
		mes "[ Nostalgic Staff ]";
		mes "Hello Guild Member.";
		next;
		mes "Here's your GuildPack. Have a nice day!";
		
		setd "$GpackID_" + .@unique_id$,1;
		#GuildPackReward = 1;
		/////GuildMember REWARD
			getitembound 33061,1,Bound_Account;
			getitembound 55741,1,Bound_Account;
			getitembound 55742,1,Bound_Account;

   			getitembound 4142,1,Bound_Account;
			getitembound 4318,2,Bound_Account;
			getitembound 55806,1,Bound_Account;						
		////
		end;
	}
no:
	mes "Well Okay";
	close;

	query_sql "CREATE TABLE IF NOT EXISTS `data_gpack` ( `ID` int(11) unsigned NOT NULL DEFAULT '0', `GID` int(11) NOT NULL, PRIMARY KEY (`ID`) ) ENGINE=MyISAM";
	query_sql "ALTER TABLE `data_gpack` MODIFY `ID` int(11) NOT NULL AUTO_INCREMENT";
