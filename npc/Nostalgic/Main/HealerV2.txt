//= Healer (Non-VIP basic | VIP full buffs | Repair+Identify for ALL)
//===== By: ==================================================

//===== Description: =========================================
//= Non-VIP: Heal + Bless + Agi (10 min)
//= VIP: Full Buff Package (20 min) (no Kaahi/Kaizel/etc)
//= ALL: RepairAll + IdentifyAll
//============================================================

-	script	Healer	-1,{

	.@Price = 0;	// Zeny required for heal
	.@Buffs = 1;	// Also buff players? (1: yes / 0: no)
	.@Delay = 0;	// Heal delay, in seconds

	.@t_nonvip = 600000;   // 10 min (ms)
	.@t_vip    = 1200000;  // 20 min (ms)

	if (@HD > gettimetick(2))
		end;

	if (.@Price) {
		message strcharinfo(0), "Healing costs " + callfunc("F_InsertComma",.@Price) + " Zeny.";
		if (Zeny < .@Price)
			end;
		if (select("^0055FFHeal^000000:^777777Cancel^000000") == 2)
			end;
		Zeny -= .@Price;
	}

	// Heal for ALL
	specialeffect2 EF_HEAL2;
	percentheal 100,100;

	// Repair + Identify for ALL (VIP or not)
	repairall;
	identifyall;

	if (.@Buffs) {

		// Remove common bad status
		sc_end SC_STONE;
		sc_end SC_FREEZE;
		sc_end SC_SLEEP;
		sc_end SC_CURSE;
		sc_end SC_SILENCE;
		sc_end SC_CONFUSION;
		sc_end SC_BLIND;
		sc_end SC_BLEEDING;
		sc_end SC_DECREASEAGI;
		sc_end SC_POISON;
		sc_end SC_HALLUCINATION;
		sc_end SC_STRIPWEAPON;
		sc_end SC_STRIPARMOR;
		sc_end SC_STRIPHELM;
		sc_end SC_STRIPSHIELD;
		sc_end SC_BERSERK;

		// -------------------------
		// Non-VIP: Bless + Agi (10 min)
		// -------------------------
		specialeffect2 EF_INCAGILITY;
		sc_start SC_INCREASEAGI, .@t_nonvip, 10;
		specialeffect2 EF_BLESSING;
		sc_start SC_BLESSING, .@t_nonvip, 10;

		// -------------------------
		// VIP: Full Buff Package (20 min)
		// -------------------------
		if (vip_status(1)) {

			// Extend Bless/Agi duration to VIP time
			sc_start SC_INCREASEAGI, .@t_vip, 10;
			sc_start SC_BLESSING, .@t_vip, 10;

			// Chemical Protection set
			sc_start SC_CP_WEAPON, .@t_vip, 10;
			sc_start SC_CP_ARMOR, .@t_vip, 10;
			sc_start SC_CP_SHIELD, .@t_vip, 10;
			sc_start SC_CP_HELM, .@t_vip, 10;

			// Full buffs (no Kaahi/Kaizel/Kaite)
			skilleffect 74,0;  sc_start SC_MAGNIFICAT, .@t_vip, 5;
			skilleffect 75,0;  sc_start SC_GLORIA, .@t_vip, 5;
			skilleffect 66,0;  sc_start SC_IMPOSITIO, .@t_vip, 1;
			skilleffect 67,0;  sc_start SC_SUFFRAGIUM, .@t_vip, 3;
			skilleffect 8,0;   sc_start SC_ENDURE, .@t_vip, 10;

			skilleffect 383,0; sc_start SC_WINDWALK, .@t_vip, 5;
			skilleffect 157,0; sc_start SC_ENERGYCOAT, .@t_vip, 1;
			skilleffect 380,0; sc_start SC_TRUESIGHT, .@t_vip, 5;

			skilleffect 114,0; sc_start SC_MAXIMIZEPOWER, .@t_vip, 1;
			skilleffect 459,0; sc_start SC_ADRENALINE2, .@t_vip, 1;

			skilleffect 355,0; sc_start SC_AURABLADE, .@t_vip, 5;
			skilleffect 357,0; sc_start SC_CONCENTRATION, .@t_vip, 1;
			skilleffect 155,0; sc_start SC_LOUD, .@t_vip, 1;

			sc_start SC_ASPDPOTION2, .@t_vip, 0;

			// Food buffs
			sc_start SC_STRFOOD, .@t_vip, 10;
			sc_start SC_AGIFOOD, .@t_vip, 10;
			sc_start SC_VITFOOD, .@t_vip, 10;
			sc_start SC_INTFOOD, .@t_vip, 10;
			sc_start SC_DEXFOOD, .@t_vip, 10;
			sc_start SC_LUKFOOD, .@t_vip, 10;
			sc_start SC_HITFOOD, .@t_vip, 30;
			sc_start SC_FLEEFOOD, .@t_vip, 30;
			sc_start SC_BATKFOOD, .@t_vip, 10;
			sc_start SC_MATKFOOD, .@t_vip, 10;

			// Spirit per job (VIP time)
			.@spirit = 0;
			switch (basejob) {
				case Job_Alchemist:      .@spirit = 445; break;
				case Job_Monk:           .@spirit = 447; break;
				case Job_Star_Gladiator: .@spirit = 448; break;
				case Job_Sage:           .@spirit = 449; break;
				case Job_Crusader:       .@spirit = 450; break;
				case Job_SuperNovice:    .@spirit = 451; break;
				case Job_Knight:         .@spirit = 452; break;
				case Job_Wizard:         .@spirit = 453; break;
				case Job_Priest:         .@spirit = 454; break;
				case Job_Bard:
				case Job_Dancer:         .@spirit = 455; break;
				case Job_Rogue:          .@spirit = 456; break;
				case Job_Assassin:       .@spirit = 457; break;
				case Job_Blacksmith:     .@spirit = 458; break;
				case Job_Hunter:         .@spirit = 460; break;
				case Job_Soul_Linker:    .@spirit = 461; break;
				default:
					if (upper == 1 && baselevel < 70)
						.@spirit = 494;
					break;
			}

			if (.@spirit > 0) {
				sc_start4 SC_SPIRIT, .@t_vip, 5, .@spirit, 0, 0;
				skilleffect .@spirit, 5;
			}
		}
	}

	if (.@Delay)
		@HD = gettimetick(2) + .@Delay;
	end;
}
