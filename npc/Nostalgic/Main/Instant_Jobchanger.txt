function Manual_Menu {

	Manual_Check_Riding();
	Manual_Check_SkillPoints();

	.@eac = eaclass();
	.@rebirth_possible = Manual_Can_Rebirth();
	.@first_eac = .@eac&EAJ_BASEMASK;
	.@second_eac = .@eac&EAJ_UPPERMASK;

	deletearray .@job_opt[0], getarraysize(.@job_opt);

	if (.@rebirth_possible) {
		Manual_Require_Level(.Req_Rebirth[0], .Req_Rebirth[1]);
		Manual_Job_Options(.@job_opt, Job_Novice_High);
	}

	if (.SecondExpanded &&
		(.@eac&EAJ_UPPERMASK) == EAJ_SUPER_NOVICE &&
		!(eaclass()&EAJL_THIRD)) {
		if (!Manual_Is_Baby(.@eac)) {
			Manual_Require_Level(.Req_Exp_SNOVI[0], .Req_Exp_SNOVI[1]);
		}
	}

	if (.SecondExpanded &&
		((.@eac&(~EAJL_BABY)) == EAJ_NINJA ||
		(.@eac&(~EAJL_BABY)) == EAJ_GUNSLINGER)) {
		if (!Manual_Is_Baby(.@eac)) {
			Manual_Require_Level(.Req_Exp_NJ_GS[0], .Req_Exp_NJ_GS[1]);
		}
	}

	if (.@first_eac == EAJ_NOVICE && .@second_eac != EAJ_SUPER_NOVICE) {
		Manual_Require_Level(.Req_First[0], .Req_First[1]);

		switch(Class) {
			case Job_Novice:
				Manual_Job_Options(.@job_opt,
					Job_Swordman,
					Job_Mage,
					Job_Archer,
					Job_Acolyte,
					Job_Merchant,
					Job_Thief,
					Job_Super_Novice,
					Job_Taekwon,
					Job_Gunslinger,
					Job_Ninja
				);
				break;

			case Job_Novice_High:
				if (.LastJob && lastJob)
					Manual_Job_Options(.@job_opt, roclass((eaclass(lastJob)&EAJ_BASEMASK)|EAJL_UPPER));
				else
					Manual_Job_Options(.@job_opt,
						Job_Swordman_High,
						Job_Mage_High,
						Job_Archer_High,
						Job_Acolyte_High,
						Job_Merchant_High,
						Job_Thief_High
					);
				break;

			default:
				mes .npcname$;
				mes "An error has occurred.";
				close;
		}
	}
	else if (Manual_Is_First_Cls(.@eac) ||
			 Manual_Is_First_Cls(.@eac&(~EAJL_UPPER))) {

		.@class1 = roclass(.@eac|EAJL_2_1);
		.@class2 = roclass(.@eac|EAJL_2_2);

		if (.LastJob && lastJob && (.@eac&EAJL_UPPER)) {
			Manual_Require_Level(.Req_Second[0], .Req_Second[1]);
			Manual_Job_Options(.@job_opt, lastJob + Job_Novice_High);
		}
		else {
			if (.@class1 > 0) {
				Manual_Require_Level(.Req_Second[0], .Req_Second[1]);
				Manual_Job_Options(.@job_opt, .@class1);
			}
			if (.@class2 > 0) {
				Manual_Require_Level(.Req_Second[0], .Req_Second[1]);
				Manual_Job_Options(.@job_opt, .@class2);
			}
		}
	}

	Manual_Job_Menu(.@job_opt);
	return;
}

function Manual_Require_Level {
	if (BaseLevel < getarg(0) || JobLevel < getarg(1)) {
		.@blvl = getarg(0) - BaseLevel;
		.@jlvl = getarg(1) - JobLevel;

		mes .npcname$;
		mes "Level requirement:";
		mes ((getarg(0)>1) ? "^bb0000"+getarg(0)+"^000000 Base / " : "")+"^00bb00"+getarg(1)+"^000000 Job";
		mes "You need " +
			((.@blvl > 0) ? "^bb0000"+.@blvl+"^000000 more base levels " + ((.@jlvl > 0) ? "and " : "") : "") +
			((.@jlvl > 0) ? "^00bb00"+.@jlvl+"^000000 more job levels " : "") +
			"to continue.";
		close;
	}
	return;
}

function Manual_Is_Baby {
	return ((getarg(0, eaclass())&EAJL_BABY)>0);
}

function Manual_Can_Rebirth {
	if (!.RebirthClass)
		return false;
	if (!(eaclass()&EAJL_2))
		return false;
	if (eaclass()&(EAJL_UPPER|EAJL_THIRD))
		return false;
	if (roclass(eaclass()|EAJL_UPPER) < 0)
		return false;
	if (Manual_Is_Baby())
		return false;
	return true;
}

function Manual_Is_First_Cls {
	return (getarg(0) <= EAJ_TAEKWON);
}

function Manual_Check_Riding {
	if (checkfalcon() || checkcart() || checkriding() || ismounting()) {
		mes .npcname$;
		mes "Please remove your mount, cart, falcon, or Peco before proceeding.";
		close;
	}
	return;
}

function Manual_Check_SkillPoints {
	if (.SkillPointCheck && SkillPoint) {
		mes .npcname$;
		mes "Please use all your skill points before proceeding.";
		close;
	}
	return;
}

function Manual_Job_Options {
	.@argcount = getargcount();
	.@arr_size = getarraysize(getarg(0));

	for (.@i = 1; .@i < .@argcount; .@i++) {
		setarray getelementofarray(getarg(0), .@arr_size++), getarg(.@i);
	}
	return;
}

function Manual_Job_Menu {
	while(true) {
		.@opt_cnt = getarraysize(getarg(0));

		if (.@opt_cnt <= 0) {
			mes .npcname$;
			mes "No more jobs are available.";
			close;
		}

		.@selected = 0;

		if (.@opt_cnt > 1) {
			mes .npcname$;
			mes "Select a job.";
			.@menu$ = "";

			for (.@i = 0; .@i < .@opt_cnt; .@i++) {
				if (getelementofarray(getarg(0), .@i) == Job_Novice_High)
					.@jobname$ = "^0055FFRebirth^000000";
				else
					.@jobname$ = jobname(getelementofarray(getarg(0), .@i));

				.@menu$ = .@menu$ + " ~ " + .@jobname$ + ":";
			}

			.@menu$ = .@menu$ + " ~ ^777777Cancel^000000";
			.@selected = select(.@menu$) - 1;

			if (.@selected < 0 || .@selected >= .@opt_cnt)
				close;

			next;
		}

		.@class = getelementofarray(getarg(0), .@selected);

		if (.@class == Job_Super_Novice && BaseLevel < .SNovice) {
			mes .npcname$;
			mes "A base level of "+.SNovice+" is required to turn into a "+jobname(.@class)+".";
			close;
		}

		Manual_Confirm_Change(.@class, .@opt_cnt > 1);
		next;
	}
	return;
}

function Manual_Confirm_Change {
	.@class = getarg(0, -1);
	.@back = getarg(1, false);

	if (.@class < 0 || eaclass(.@class) == -1) {
		mes .npcname$;
		mes "Unknown Class Error.";
		close;
	}

	mes .npcname$;
	mes "Do you want to change into ^0055FF"+jobname(.@class)+"^000000 class?";

	.@job_option$ = " ~ Change into ^0055FF"+jobname(.@class)+"^000000 class";

	if (.@class == Job_Novice_High)
		.@job_option$ = " ~ ^0055FFRebirth^000000";

	if (select(.@job_option$+": ~ ^777777"+((.@back) ? "Go back" : "Cancel")+"^000000") == 1) {
		Manual_Job_Change(.@class);
	}

	if (!.@back)
		close;

	return;
}

function Manual_Job_Change {
	.@previous_class = Class;
	.@to_cls = getarg(0);

	next;
	mes .npcname$;
	mes "You are now "+callfunc("F_InsertArticle", jobname(.@to_cls))+"!";

	if (.@to_cls == Job_Novice_High && .LastJob)
		lastJob = Class;

	jobchange .@to_cls;

	if (.@to_cls == Job_Novice_High)
		resetlvl(1);

	specialeffect2 EF_ANGEL2;
	specialeffect2 EF_ELECTRIC;

	if (.@previous_class != Class) {
		if (.Platinum)
			callfunc "F_GetPlatinumSkills";
	}

	close;
}

	end;
