valmorra,182,128,4	script	Private MVP Room	11008,{
	mes "[Private MVP Room]";
	mes "Please select a room range.";
	next;

	.@range = select("Rooms 1-50:Rooms 51-100:Rooms 101-150:Rooms 151-200:Rooms 201-250:Rooms 251-298");
	.@start = ( .@range == 1 ) ? 1 : ( .@range == 2 ) ? 51 : ( .@range == 3 ) ? 101 : ( .@range == 4 ) ? 151 : ( .@range == 5 ) ? 201 : 251;
	.@end   = ( .@range == 1 ) ? 50 : ( .@range == 2 ) ? 100 : ( .@range == 3 ) ? 150 : ( .@range == 4 ) ? 200 : ( .@range == 5 ) ? 250 : 298;

	.@menu$ = "";
	for ( .@i = .@start; .@i <= .@end; .@i++ ) {
		.@check_inst = getvariableofnpc(.room_instance_id[.@i], "MVP Summoner Master Core");
		.@status$ = "Available";
		
		// Use instance_mapname to safely verify if the instance is still active
		if ( .@check_inst > 0 && instance_mapname("06guild_01", .@check_inst) != "" ) {
			.@status$ = getvariableofnpc(.color$[getvariableofnpc(.room_owner_type[.@i], "MVP Summoner Master Core")], "MVP Summoner Master Core") + getvariableofnpc(.room_owner_name$[.@i], "MVP Summoner Master Core") +"^000000";
		}
		.@menu$ = .@menu$ + "Room #"+ .@i +" ["+ .@status$ +"]:";
	}
	
	.@room = .@start + select(.@menu$) - 1;
	.@inst_id = getvariableofnpc(.room_instance_id[.@room], "MVP Summoner Master Core");

	if ( .@inst_id > 0 && instance_mapname("06guild_01", .@inst_id) != "" ) {
		.@owner_type = getvariableofnpc(.room_owner_type[.@room], "MVP Summoner Master Core");
		.@owner_id   = getvariableofnpc(.room_owner_id[.@room], "MVP Summoner Master Core");
		if ( getcharid(.@owner_type) == .@owner_id ) {
			.@map_name$ = instance_mapname("06guild_01", .@inst_id);
			if (.@map_name$ != "") { warp .@map_name$, 49, 49; close; }
		}
		mes "[Private MVP Room]";
		mes "This room is currently reserved or expired.";
		close;
	}

	mes "[Private MVP Room]";
	mes "Reserve this room slot for...";
	next;
	.@choice = select( "For my party members:For my guild members:For personal account use" );
	
	// Map selection directly to native rAthena Engine string types
	.@mode$ = ( .@choice == 1 ) ? "Party" : ( .@choice == 2 ) ? "Guild" : "Char";
	.@type = .@choice; // Keep original for owner tracking

	if ( ( .@choice == 1 && !getcharid(1) ) || ( .@choice == 2 && !getcharid(2) ) || Zeny < 10000000 ) {
		mes "[Private MVP Room]";
		mes "Requirement not met (Party/Guild missing or Insufficient Zeny).";
		close;
	}

	for ( .@i = 1; .@i <= 298; .@i++ ) {
		.@check_id = getvariableofnpc(.room_instance_id[.@i], "MVP Summoner Master Core");
		if ( .@check_id > 0 && instance_mapname("06guild_01", .@check_id) != "" ) {
			if ( getvariableofnpc(.room_owner_type[.@i], "MVP Summoner Master Core") == .@type && getvariableofnpc(.room_owner_id[.@i], "MVP Summoner Master Core") == getcharid(.@type) ) {
				mes "[Private MVP Room]";
				mes "Your group already holds a reservation on Room #"+ .@i +".";
				close;
			}
		}
	}

	Zeny -= 10000000;
	.@new_inst = instance_create("MVP_Room", .@mode$);
	if ( .@new_inst <= 0 ) { mes "Error: Instance creation failed. (Code: "+ .@new_inst +")"; Zeny += 10000000; close; }

	set getvariableofnpc(.room_timeout_duration[.@room], "MVP Summoner Master Core"), 60;
	set getvariableofnpc(.room_instance_id[.@room], "MVP Summoner Master Core"), .@new_inst;
	set getvariableofnpc(.room_owner_type[.@room], "MVP Summoner Master Core"), .@type;
	set getvariableofnpc(.room_owner_id[.@room], "MVP Summoner Master Core"), getcharid(.@type);
	set getvariableofnpc(.room_owner_name$[.@room], "MVP Summoner Master Core"), strcharinfo( ( .@type == 3 )? 0 : .@type );
	set getvariableofnpc(.room_start_tick[.@room], "MVP Summoner Master Core"), gettimetick(2);

	.@target_map$ = instance_mapname("06guild_01", .@new_inst);
	warp .@target_map$, 49, 49;
	close;
}

06guild_01,49,49,4	script	MVP Summoner Master Core	67,{
	if ( !playerattached() ) end;
	.@inst_id = instance_id();
	if ( .@inst_id <= 0 ) end;

	.@room = 0;
	for ( .@i = 1; .@i <= 298; .@i++ ) {
		if ( .room_instance_id[.@i] == .@inst_id ) { .@room = .@i; break; }
	}

	if ( !.@room ) { mes "[MVP Summoner]"; mes "Error: Could not identify room context."; close; }

	.@elapsed = gettimetick(2) - .room_start_tick[.@room];
	.@left = (getvariableofnpc(.room_timeout_duration[.@room], "MVP Summoner Master Core") * 60) - .@elapsed;
	if (.@left < 0) .@left = 0;

	mes "[MVP Summoner]";
	mes sprintf("Time remaining: %dh %02dm %02ds", .@left / 3600, (.@left % 3600) / 60, .@left % 60);
	mes "Hi there! How can I assist your party inside the room today?";
	next;

	switch ( select(.menu$) ) {
	case 1: // Re-summon
		if ( mobcount("this", "MVP Summoner Master Core::OnMobDead") ) { mes "There are still active monsters."; close; }
		if ( !#last_mvp_id || !#last_mvp_amount ) { mes "No previous parameters saved."; close; }
		.@totalcost = .mvpcost * #last_mvp_amount;
		if ( Zeny < .@totalcost ) { mes "Not enough zeny."; close; }
		mes "Re-summoning previous targets now."; close2;
		Zeny -= .@totalcost;
		monster "this", 0, 0, "--ja--", #last_mvp_id, #last_mvp_amount, "MVP Summoner Master Core::OnMobDead";
		end;
	case 2: // Summon MVP
		if ( mobcount("this", "MVP Summoner Master Core::OnMobDead") ) { mes "Active monsters exist."; close; }
		set .@menu, select(.mvpid_menu$) - 1;
		mes "How many instances?"; input .@amount;
		if ( .@amount < 1 || .@amount > 10 ) { mes "Invalid quantity (1-10)."; close; }
		.@totalcost = .mvpcost * .@amount;
		if ( Zeny < .@totalcost ) { mes "Insufficient zeny."; close; }
		#last_mvp_id = .mvpid[.@menu]; #last_mvp_amount = .@amount;
		close2; Zeny -= .@totalcost;
		monster "this", 0, 0, "--ja--", .mvpid[.@menu], .@amount, "MVP Summoner Master Core::OnMobDead";
		end;
	case 3: // Summon Mini-boss
		if ( mobcount("this", "MVP Summoner Master Core::OnMobDead") ) { mes "Active monsters exist."; close; }
		set .@menu, select(.bossid_menu$) - 1;
		if ( Zeny < .bosscost ) { mes "Insufficient zeny."; close; }
		close2; Zeny -= .bosscost;
		monster "this", 0, 0, "--ja--", .bossid[.@menu], 1, "MVP Summoner Master Core::OnMobDead";
		end;
	case 4: // Heal
		sc_end SC_STONE; sc_end SC_SLOWDOWN; sc_end SC_FREEZE; sc_end SC_SLEEP; sc_end SC_CURSE; sc_end SC_SILENCE; sc_end SC_CONFUSION; sc_end SC_BLIND; sc_end SC_BLEEDING; sc_end SC_DECREASEAGI; sc_end SC_POISON; sc_end SC_HALLUCINATION; sc_end SC_STRIPWEAPON; sc_end SC_STRIPARMOR; sc_end SC_STRIPHELM; sc_end SC_STRIPSHIELD; sc_end SC_CHANGEUNDEAD; sc_end SC_ORCISH; sc_end SC_BERSERK; sc_end SC_SKE; sc_end SC_SWOO; sc_end SC_SKA;
		percentheal 100,100;
		sc_start SC_BLESSING,600000,10; sc_start SC_INCREASEAGI,600000,10;
		specialeffect2 EF_HEAL; specialeffect2 EF_BLESSING; specialeffect2 EF_INCAGILITY;
		mes "Fully healed and buffed!"; close;
	case 5: // Leave
		warp .respawnmap$, .respawnx, .respawny; close;
	case 6: // Give up
		mes "Are you sure? All players will be kicked.";
		next;
		if ( select("Yes:No") == 1 ) {
			instance_warpall .respawnmap$, .respawnx, .respawny;
			instance_destroy(.@inst_id);
		}
		end;
	}
	close;

OnMobDead:
	if ( !playerattached() ) end;
	callfunc "F_AddMvpRankPoint";
	callfunc "F_WorldBoss_AddMvpKill", 1;
	end;

OnInit:
	.timeout = 60; .rentcost = 10000000; .mvpcost = 5000000; .bosscost = 2000000;
	.respawnmap$ = "valmorra"; .respawnx = 171; .respawny = 127;
	.color$[1] = "^EE8800"; .color$[2] = "^70CC11"; .color$[3] = "^0000FF";
	setarray .mvpid[0], 1511,1785,1630,1039,1874,1272,1719,1046,1389,1112,1115,1418,1871,1252,1768,1086,1885,1832,1492,1251,1779,1688,1373,1147,1059,1150,1087,1190,1038,2022,1157,1159,1623,1583,1312,1751,1685,1917,1658;
	setarray .bossid[0], 1096,1388,1795,1830,1839,1309,1283,1302,1198,1582,1091,1093,1205,1783,1592,1120,1259,1720,1090,1289,1262,1203,1870,1295,1829,1204,1089,1092,1765;
	.menu$ = "Re-Summon Previous MVP:Summon MVP:Summon Mini-boss:Heal:Leave this room:Give up this room";
	.@size = getarraysize(.mvpid);
	for ( .@i = 0; .@i < .@size; .@i++ ) .mvpid_menu$ = .mvpid_menu$ + getmonsterinfo(.mvpid[.@i], MOB_NAME) + ":";
	.@size = getarraysize(.bossid);
	for ( .@i = 0; .@i < .@size; .@i++ ) .bossid_menu$ = .bossid_menu$ + getmonsterinfo(.bossid[.@i], MOB_NAME) + ":";
	end;
}
