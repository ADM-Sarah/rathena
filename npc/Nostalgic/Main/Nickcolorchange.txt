//============================================================
// Nostalgic RO - Color Nick Box Functions
//============================================================

function	script	F_ColorNickEffect	{
	specialeffect2 568;
	return;
}

function	script	F_WoEColorNickBox	{
	if (rand(1,100) <= 10)
		.@color = 0;
	else
		.@color = rand(1,40);

	atcommand "@colornick " + .@color;
	callfunc "F_ColorNickEffect";

	if (.@color == 0)
		dispbottom "[WoE Color Box] Your nickname color has been reset to default.";
	else
		dispbottom "[WoE Color Box] Your nickname color changed to Color No. " + .@color + ".";

	end;
}
