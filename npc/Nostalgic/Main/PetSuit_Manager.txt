//============================================================
// Pet & Suit Manager
// Final stable version
// Requirement:
// - 10,000,000 Zeny
// - 100 PVP Points
// - Gold Credit (33019) x1
// - Jellopy (969) x10000
//============================================================

valmorra,151,214,4	script	Suit & Pet Manager	11011,{

OnClick:
	goto OnMenu;

OnInit:
	freeloop(1);

	// -----------------------------------------------------------
	// Basic shop settings
	// -----------------------------------------------------------
	.Announce = 0;		// 1 = announce trade completion
	.ShowSlot = 1;		// 2 = all equip / 1 = only if slot > 0 / 0 = never
	.ShowID = 0;  		// 1 = show item IDs
	.ShowZeny = 0;		// 1 = show Zeny cost in shop list
	.MaxStack = 1;		// max purchase at one time
	.ignore_secure_npctimeout = 1;

	// -----------------------------------------------------------
	// Dialog visual settings
	// -----------------------------------------------------------
	.DialogW = 360;
	.DialogH = 300;

	// -----------------------------------------------------------
	// Optional points variable
	// -----------------------------------------------------------
	setarray .Points$[0],"#PVP_KillPoints","PVP Points";

	// -----------------------------------------------------------
	// Shop categories
	// -----------------------------------------------------------
	setarray .Shops$[1],"~ Custom Suit","~ Custom Pet";

	// -----------------------------------------------------------
	// Add(<shop ID>,<reward ID>,<reward amount>,
	//     <Zeny cost>,<point cost>,
	//     <required item ID>,<required item amount>{,...});
	// -----------------------------------------------------------

	// Custom Suit
	callsub S_Add,1,85513,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85514,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85515,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85516,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85517,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85518,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85519,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85520,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85521,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85522,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85523,1,10000000,100,33019,1,969,10000;
	callsub S_Add,1,85524,1,10000000,100,33019,1,969,10000;

	// Custom Pet
	callsub S_Add,2,9057,1,10000000,100,33019,1,969,10000;
	callsub S_Add,2,9058,1,10000000,100,33019,1,969,10000;
	callsub S_Add,2,9059,1,10000000,100,33019,1,969,10000;
	callsub S_Add,2,9060,1,10000000,100,33019,1,969,10000;
	callsub S_Add,2,9061,1,10000000,100,33019,1,969,10000;
	callsub S_Add,2,9062,1,10000000,100,33019,1,969,10000;

	freeloop(0);

	.menu$ = "";
	for (.@i = 1; .@i <= getarraysize(.Shops$); .@i++) {
		.menu$ += .Shops$[.@i] + ":";
		npcshopdelitem "petsuitshop"+.@i,909;
	}
	end;

OnMenu:
	if (.ignore_secure_npctimeout)
		ignoretimeout 1;

	setdialogsize(.DialogW,.DialogH);
	setdialogalign(DIALOG_ALIGN_CENTER);

	mes "<b>^000000[ ^FF7F00Pet & Suit Manager^000000 ]</b>";
	mes "^000000Welcome to the ^FF7F00Premium Pet & Suit Exchange^000000.";
	mes "^000000This shop offers exclusive ^1E90FFCustom Suits^000000 and ^1E90FFSpecial Pets^000000.";
	mes "^000000All items in this shop utilize the ^FF7F00PVP Point System^000000 along with additional item requirements.^000000";
	mes "^000000Your current ^0055FF"+.Points$[1]+"^000000:^000000 ^1E90FF"+getd(.Points$[0])+"^000000";
	mes "^000000Please ensure you have sufficient requirements before proceeding.^000000";
	mes callsub(S_Line);
	next;

	.@size = getarraysize(@i);
	if (!.@size)
		@shop_index = select(.menu$);
	else if (.@size == 1)
		@shop_index = @i[0];
	else {
		.@menu$ = "";
		for (.@j = 0; .@j < .@size; .@j++)
			.@menu$ += .Shops$[@i[.@j]] + ":";
		@shop_index = @i[select(.@menu$)-1];
	}
	deletearray @i[0],getarraysize(@i);

	if (.Shops$[@shop_index] == "") {
		message strcharinfo(0),"An error has occurred.";
		end;
	}

	dispbottom "Your current PVP Points: " + getd(.Points$[0]);
	dispbottom "Select one item at a time.";

	close2;
	callshop "petsuitshop"+@shop_index,1;
	npcshopattach "petsuitshop"+@shop_index;
	end;

OnBuyItem:
	if (.ignore_secure_npctimeout)
		ignoretimeout 1;

	setdialogsize(.DialogW,.DialogH);
	setdialogalign(DIALOG_ALIGN_CENTER);

	// .@q[] : RewardID, BoughtAmt, RewardAmt, BaseAmt, ReqZeny, ReqPts, { ReqItem, ReqAmt, ... }
	setarray .@q[0],@bought_nameid[0],((@bought_quantity[0] > .MaxStack) ? .MaxStack : @bought_quantity[0]);
	copyarray .@q[3],getd(".q_"+@shop_index+"_"+.@q[0]+"[0]"),getarraysize(getd(".q_"+@shop_index+"_"+.@q[0]));
	.@q[2] = .@q[1] * .@q[3];

	if (!.@q[2] || .@q[2] > 30000) {
		message strcharinfo(0),"You can't purchase that many "+getitemname(.@q[0])+".";
		end;
	}

	mes "<b>^000000[ ^FF7F00Pet & Suit Manager^000000 ]</b>";
	mes "^000000Your current ^0055FF"+.Points$[1]+"^000000:^000000 ^1E90FF"+getd(.Points$[0])+"^000000";
	mes "^000000Shop:^000000 ^FF7F00" + .Shops$[@shop_index] + "^000000";
	mes "^000000Your Zeny:^000000 ^1E90FF" + F_InsertComma(Zeny) + "^000000";
	mes callsub(S_Line);

	mes "<b>^000000Reward:^000000</b> ^0055FF"
		+ ((.@q[2] > 1) ? .@q[2]+"x " : "")
		+ "^i[" + .@q[0] + "] "
		+ callsub(S_Slot,.@q[0])
		+ "^000000";

	mes "<b>^000000Requirements:^000000</b>";
	disable_items;

	set @qe[0],0;

	if (.@q[4])
		mes "^3CB371»^000000 " + callsub(S_Chk,Zeny,.@q[4]*.@q[1]) + F_InsertComma((.@q[4]*.@q[1])) + " Zeny^000000";

	if (.@q[5])
		mes "^3CB371»^000000 "
			+ callsub(S_Chk,getd(.Points$[0]),.@q[5]*.@q[1])
			+ (.@q[5]*.@q[1])+" "+.Points$[1]
			+ " ^777777("+getd(.Points$[0])+"/"+(.@q[5]*.@q[1])+")^000000";

	if (.@q[6]) for(.@i = 6; .@i < getarraysize(.@q); .@i += 2)
		mes "^3CB371»^000000 "
			+ callsub(S_Chk,countitem(.@q[.@i]),.@q[.@i+1]*.@q[1])
			+ ((.ShowID)?"{"+.@q[.@i]+"} ":"")
			+ "^i[" + .@q[.@i] + "] "
			+ "<ITEM>" + callsub(S_Slot,.@q[.@i]) + "<INFO>" + .@q[.@i] + "</INFO></ITEM> "
			+ "^777777(" + countitem(.@q[.@i]) + "/" + (.@q[.@i+1]*.@q[1]) + ")^000000";

	setdialogsize(.DialogW,.DialogH);
	setdialogalign(DIALOG_ALIGN_CENTER);
	next;

	setarray @qe[1], getiteminfo(.@q[0],5), getiteminfo(.@q[0],11);

	// Preview enabled for equip types with viewid
	if (@qe[2] > 0 && ((@qe[1] & 1) || (@qe[1] & 256) || (@qe[1] & 512) || (@qe[1] & 1024) || (@qe[1] & 2048) || (@qe[1] & 4096) || (@qe[1] & 4) || (@qe[1] & 8192)))
		.@preview = 1;

	addtimer 1000, strnpcinfo(3)+"::OnEnd";
	while(1) {
		switch(select(
			"^3CB371» Purchase ^0055FF"+getitemname(.@q[0])+"^000000"
			+ ":"
			+ ((.@preview && !@qe[7])?"^1E90FF» Preview Item^000000:":"")
			+ "^FF0000» Cancel^000000"
		)) {
		case 1:
			if (@qe[0]) {
				mes "<b>^000000[ ^FF7F00Pet & Suit Manager^000000 ]</b>";
				mes "^FF0000You're missing one or more requirements.^000000";
				close;
			}
			if (!checkweight(.@q[0],.@q[2])) {
				mes "<b>^000000[ ^FF7F00Pet & Suit Manager^000000 ]</b>";
				mes "^FF0000You need more weight capacity to complete this trade.^000000";
				close;
			}

			if (.@q[4]) Zeny -= (.@q[4]*.@q[1]);
			if (.@q[5]) setd .Points$[0], getd(.Points$[0])-(.@q[5]*.@q[1]);
			if (.@q[6]) for(.@i = 6; .@i < getarraysize(.@q); .@i += 2)
				delitem .@q[.@i], .@q[.@i+1]*.@q[1];

			getitem .@q[0], .@q[2];

			if (.Announce)
				announce strcharinfo(0)+" has created "+((.@q[2] > 1)? .@q[2]+"x "+getitemname(.@q[0]) : callfunc("F_InsertArticle",getitemname(.@q[0])))+"!",0;

			specialeffect2 EF_FLOWERLEAF;

			mes "<b>^000000[ ^FF7F00Pet & Suit Manager^000000 ]</b>";
			mes "^1E90FFTrade completed successfully!^000000";
			mes "^000000You received:^000000 ^0055FF"
				+ ((.@q[2] > 1) ? .@q[2]+"x " : "")
				+ "^i[" + .@q[0] + "] "
				+ callsub(S_Slot,.@q[0])
				+ "^000000";
			mes "^000000Your remaining ^0055FF"+.Points$[1]+"^000000:^000000 ^1E90FF"+getd(.Points$[0])+"^000000";
			close;

		case 2:
			if (.@preview && !@qe[7]) {
				setarray @qe[3], getlook(LOOK_HEAD_BOTTOM), getlook(LOOK_HEAD_TOP), getlook(LOOK_HEAD_MID), getlook(LOOK_ROBE), 1;
				if ((@qe[1] & 1) || (@qe[1] & 4096)) changelook LOOK_HEAD_BOTTOM, @qe[2];
				else if ((@qe[1] & 256) || (@qe[1] & 1024)) changelook LOOK_HEAD_TOP, @qe[2];
				else if ((@qe[1] & 512) || (@qe[1] & 2048)) changelook LOOK_HEAD_MID, @qe[2];
				else if ((@qe[1] & 4) || (@qe[1] & 8192)) changelook LOOK_ROBE, @qe[2];
				break;
			}
			close;

		case 3:
			close;
		}
	}

OnEnd:
	if (.ignore_secure_npctimeout)
		ignoretimeout 0;

	if (@qe[7]) {
		changelook LOOK_HEAD_BOTTOM, @qe[3];
		changelook LOOK_HEAD_TOP, @qe[4];
		changelook LOOK_HEAD_MID, @qe[5];
		changelook LOOK_ROBE, @qe[6];
	}
	deletearray @qe[0],8;
	end;

//------------------------------------------------------------
// Add shop item data
//------------------------------------------------------------
S_Add:
	if (getitemname(getarg(1)) == "null") {
		debugmes "Quest reward #"+getarg(1)+" invalid (skipped).";
		return;
	}

	setarray .@j[0],getarg(2),getarg(3),getarg(4);

	for(.@i = 5; .@i < getargcount(); .@i += 2) {
		if (getitemname(getarg(.@i)) == "null") {
			debugmes "Quest requirement #"+getarg(.@i)+" invalid (skipped).";
			return;
		}
		setarray .@j[.@i-2], getarg(.@i), getarg(.@i+1);
	}

	copyarray getd(".q_"+getarg(0)+"_"+getarg(1)+"[0]"), .@j[0], getarraysize(.@j);
	npcshopadditem "petsuitshop"+getarg(0), getarg(1), ((.ShowZeny) ? getarg(3) : 0);
	return;

//------------------------------------------------------------
// Requirement color checker
//------------------------------------------------------------
S_Chk:
	if (getarg(0) < getarg(1)) {
		@qe[0] = 1;
		return "^FF0000";
	}
	return "^00BFFF";

//------------------------------------------------------------
// Slot display helper
//------------------------------------------------------------
S_Slot:
	.@s$ = getitemname(getarg(0));
	switch(.ShowSlot) {
		case 1:
			if (!getitemslots(getarg(0))) return .@s$;
		case 2:
			if (getiteminfo(getarg(0),2) == 4 || getiteminfo(getarg(0),2) == 5)
				return .@s$+" ["+getitemslots(getarg(0))+"]";
		default:
			return .@s$;
	}

S_Line:
	return "^777777--------------------------------------------------^000000";
}

-	shop	petsuitshop2	-1,909:-1
