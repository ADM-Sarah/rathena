function	script	F_IsPvpMap	{
	.@m$ = getarg(0);
	if (
		.@m$ == "izlude2" ||
		.@m$ == "guild_vs1" ||
		.@m$ == "guild_vs2" ||
		.@m$ == "guild_vs3" ||
		.@m$ == "guild_vs4" ||
		.@m$ == "guild_vs5"
	) return 1;
	return 0;
}

S_CheckPvpAbuseAllRooms:

		return;

		"guild_vs3",
		"guild_vs4",
		"guild_vs5";

	for (.@m = 0; .@m < getarraysize(.@maps$); .@m++) {

		// Clear array explicitly to avoid leaks between map iterations
		cleararray .@aid[0], 0, 128; 
		.@count = getmapunits(BL_PC, .@maps$[.@m], .@aid);

		for (.@i = 0; .@i < .@count; .@i++) {

			// Skip if it's the player trying to warp
			if (.@aid[.@i] == .@myrid)
				continue;

			// Try attaching to the target player. If it fails, skip safely.
			if (!attachrid(.@aid[.@i]))
				continue;

			.@target_ip$ = getcharip();
			.@target_uid$ = #PVP_DeviceID$;
			.@target_name$ = strcharinfo(0);

			// Always detach safely
			detachrid;

			// Re-attach back to the original warping player using the stored variable
			if (!attachrid(.@myrid)) {
				// Safety fallback: if original player disconnected mid-check, stop script execution
				end; 
			}

			if (.@target_ip$ == .@myip$) {
				mes "[PVP Warper]";
				mes "^FF0000Access denied.^000000";
				mes "Someone from your IP is already";
				mes "inside a PVP room.";
				mes "";
				mes "Detected Player:";
				mes "^0000FF" + .@target_name$ + "^000000";
				close;
			}

			if (.@myuid$ != "" && .@target_uid$ != "" && .@target_uid$ == .@myuid$) {
				mes "[PVP Warper]";
				mes "^FF0000Access denied.^000000";
				mes "Someone from your device is already";
				mes "inside a PVP room.";
				mes "";
				mes "Detected Player:";
				mes "^0000FF" + .@target_name$ + "^000000";
				close;
			}
		}
	}

	return;

//============================================================
// PVP CONTROLLER
//============================================================
-	script	PVP_CONTROLLER	-1,{

OnInit:
	.SameVictimLimit = 6;
	.SameVictimTime = 60;
	end;

OnPCDieEvent:
	.@victim_map$ = strcharinfo(3);

	if (!callfunc("F_IsPvpMap", .@victim_map$))
		end;

	.@victim_aid = getcharid(3);
	.@victim_name$ = strcharinfo(0);
	.@killer_rid = killerrid;

	specialeffect2 EF_RESURRECTION;
	skilleffect "ALL_RESURRECTION", 1;
	warp "valmorra", 171, 132;

	if (.@killer_rid <= 0)
		end;

	if (!attachrid(.@killer_rid))
		end;

	if (!callfunc("F_IsPvpMap", strcharinfo(3)))
		end;

	if (getgroupid() >= 99)
		end;

	if (getcharid(3) == .@victim_aid)
		end;

	.@now = gettimetick(2);

	if (@pvp_last_victim_aid == .@victim_aid && (.@now - @pvp_last_kill_time) <= .SameVictimTime) {
		@pvp_same_victim_count++;
	} else {
		@pvp_last_victim_aid = .@victim_aid;
		@pvp_same_victim_count = 1;
	}

	@pvp_last_kill_time = .@now;

	if (@pvp_same_victim_count >= .SameVictimLimit) {
		announce "[PVP System] " + strcharinfo(0) + " has been removed from PVP for repeatedly killing the same player (" + .@victim_name$ + ").", bc_all, 0xFF0000;
		dispbottom "PVP abuse detected. Stop killing the same player repeatedly.";

		@pvp_same_victim_count = 0;
		@pvp_last_victim_aid = 0;
		@pvp_last_kill_time = 0;

		warp "SavePoint",0,0;
		end;
	}

	#PVP_KillPoints += 1;
	dispbottom "PVP Kill Point +1 | Total: " + #PVP_KillPoints;
	end;
}
