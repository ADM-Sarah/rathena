function	script	RC_GenerateCode	{
	.@len = getarg(0);
	.@chars$ = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789";
	.@max = getstrlen(.@chars$);
	.@code$ = "";

	for (.@i = 0; .@i < .@len; .@i++) {
		.@r = rand(.@max);
		.@code$ += substr(.@chars$, .@r, 1);
	}
	return .@code$;
}

function	script	RC_CreateUniqueCode	{
	for (.@try = 0; .@try < 200; .@try++) {
		.@new_code$ = callfunc("RC_GenerateCode", getarg(0));
		query_sql("SELECT COUNT(*) FROM `npc_redeem_codes_shared` WHERE `code` = '" + escape_sql(.@new_code$) + "'", .@count);
		if (.@count == 0)
			return .@new_code$;
	}
	return "";
}

function	script	RC_GetExpireText	{
	.@code_id = getarg(0);
	query_sql("SELECT IFNULL(DATE_FORMAT(`expire_at`, '%Y-%m-%d %H:%i:%s'), 'No Expiry') FROM `npc_redeem_codes_shared` WHERE `id` = " + .@code_id, .@txt$);
	return .@txt$;
}

			mes "> " + .@v_amount[.@j] + "x " + getitemname(.@v_item_id[.@j]);
			if ((.@j + 1) % 6 == 0 && .@j < .@r2 - 1) {
				next;
				mes "[Reward Items]";
			}
		}
	}
	next;

	switch(select("Back to Code List:Back to GM Menu:Cancel")) {
	case 1:
		goto GM_View;
	case 2:
		goto GM_Menu;
	default:
		close;
	}

GM_Delete:
	mes "[Redeem Code]";
	mes "Delete code by ID.";
	next;

	mes "Enter code ID.";
	input .@del_id;

	if (.@del_id <= 0) {
		mes "Invalid ID.";
		close;
	}

	query_sql(
		"SELECT `code`,`is_deleted` FROM `npc_redeem_codes_shared` WHERE `id` = " + .@del_id,
		.@del_code$, .@del_deleted
	);

	if (.@del_code$ == "") {
		mes "Code ID not found.";
		close;
	}
	if (.@del_deleted) {
		mes "This code is already deleted.";
		close;
	}

	mes "Delete this code?";
	mes "^FF0000" + .@del_code$ + "^000000";
	next;

	switch(select("Yes:No")) {
	case 1:
		break;
	default:
		close;
	}

	query_sql("UPDATE `npc_redeem_codes_shared` SET `is_deleted` = 1 WHERE `id` = " + .@del_id);
	mes "Code deleted.";
	close;

GM_Logs:
	.@log_per_page = 8;
	.@log_offset = .@log_page * .@log_per_page;

	query_sql("SELECT COUNT(*) FROM `npc_redeem_codes_shared_claims`", .@total_logs);

	if (.@total_logs <= 0) {
		mes "[Redeem Code]";
		mes "No logs found.";
		next;
		goto GM_Menu;
	}

	.@log_max_page = (.@total_logs - 1) / .@log_per_page;

	if (.@log_page < 0)
		.@log_page = 0;
	if (.@log_page > .@log_max_page)
		.@log_page = .@log_max_page;

	.@log_offset = .@log_page * .@log_per_page;

	.@rows = query_sql(
		"SELECT `code`,`account_id`,`char_id`,`char_name`,`ip`,`unique_id`,DATE_FORMAT(`claimed_at`, '%Y-%m-%d %H:%i:%s') " +
		"FROM `npc_redeem_codes_shared_claims` " +
		"ORDER BY `id` DESC LIMIT " + .@log_offset + "," + .@log_per_page,
		.@l_code$, .@l_aid, .@l_cid, .@l_name$, .@l_ip$, .@l_uid$, .@l_time$
	);

	mes "[Redeem Logs]";
	mes "Page: " + (.@log_page + 1) + " / " + (.@log_max_page + 1);
	next;

	for (.@i = 0; .@i < .@rows; .@i++) {
		mes "^FF0000Code:^000000 " + .@l_code$[.@i];
		mes "By: " + .@l_name$[.@i];
		mes "Account ID: " + .@l_aid[.@i];
		mes "Char ID: " + .@l_cid[.@i];
		mes "IP: " + .@l_ip$[.@i];
		mes "Unique ID: " + .@l_uid$[.@i];
		mes "At: " + .@l_time$[.@i];

		if ((.@i + 1) % 4 == 0 && .@i < .@rows - 1) {
			next;
			mes "[Redeem Logs]";
		}
	}
	next;

	.@log_menu$ = "";

	if (.@log_page > 0)
		.@log_menu$ += "Previous Page:";

	if (.@log_page < .@log_max_page)
		.@log_menu$ += "Next Page:";

	.@log_menu$ += "Back to GM Menu:Cancel";

	.@log_sel = select(.@log_menu$);
	.@log_extra = .@log_sel;

	if (.@log_page > 0) {
		if (.@log_extra == 1) {
			.@log_page--;
			goto GM_Logs;
		}
		.@log_extra--;
	}

	if (.@log_page < .@log_max_page) {
		if (.@log_extra == 1) {
			.@log_page++;
			goto GM_Logs;
		}
		.@log_extra--;
	}

	if (.@log_extra == 1)
		goto GM_Menu;

	close;

OnInit:
	.AnnounceSuccess = 1;
	.CodeLength = 10;
	.MaxItemsPerCode = 10;

	end;
}
