function	script	convertToDHMS	{
	.@ms = getarg(0);
	if (.@ms <= 0)
		return "^FF000000:00:00:00^000000";

	.@totalSeconds = .@ms / 1000;
	.@days    = .@totalSeconds / 86400;
	.@hours   = (.@totalSeconds % 86400) / 3600;
	.@minutes = (.@totalSeconds % 3600) / 60;
	.@seconds = .@totalSeconds % 60;

	return "^008000" + sprintf("%02d:%02d:%02d:%02d",
		.@days, .@hours, .@minutes, .@seconds) + "^000000";
}
