//============================================================
// NostalgicRO MVP Card Deposit System
// Permanent Bonus Version
// Main NPC Script
//============================================================

payon,182,294,4	script	Card Deposit Agent	109,{

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "Welcome, ^00AA00" + strcharinfo(0) + "^000000!";
	mes "Deposit MVP cards to unlock permanent tier bonuses.";
	callsub S_ApplyBonus;
	next;

	switch(select(
		"^FF0000System Information^000000",
		"^3366CCCard Collections^000000",
		"^993399My Deposited Cards^000000",
		"^00AA00Deposit Cards^000000",
		"^FF6600Withdraw Cards^000000",
		"^999999Cancel^000000"
	)) {
		case 1: callsub S_Info; close;
		case 2: callsub S_Collections; close;
		case 3: callsub S_DepositedList; close;
		case 4: callsub S_DepositMenu; close;
		case 5: callsub S_WithdrawMenu; close;
		default: close;
	}
	end;

S_Info:
	mes "[^3366CCCard Deposit Agent^000000]";
	mes "^3366CC[Normal MVP Tier]^000000";
	mes "Requirement: ^00AA99100 cards^000000 each Normal MVP card.";
	mes "";
	mes "Bonus:";
	mes "- All Stats +5";
	mes "- Reflect Magic Attack +2%";
	mes "- Reduce Long Range Damage +2%";
	mes "- Weight +200";
	next;

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "^FF0000[Rare MVP Tier]^000000";
	mes "Requirement: ^00AA9910 cards^000000 each Rare MVP card.";
	mes "";
	mes "Bonus:";
	mes "- ATK +30";
	mes "- HIT +100";
	mes "- Element Resist except Neutral +3%";
	mes "- Boss Damage +10%";
	mes "- Boss Resist +10%";
	mes "- Drop Rate +3%";
	mes "- Weight +500";
	next;

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "^FF0000Important:^000000";
	mes "- Bonus is active only while the tier is complete.";
	mes "- If you withdraw cards and tier becomes incomplete, bonus will be removed.";
	mes "- Reward can only be claimed once per Account / IP / Gepard Unique ID.";
	return;

S_Collections:
	callsub S_ApplyBonus;

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "^3366CC[Normal MVP Card Tier]^000000";
	callsub S_ShowTier, 1;
	next;

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "^FF0000[Rare MVP Card Tier]^000000";
	callsub S_ShowTier, 2;
	return;

S_DepositedList:
	callsub S_ApplyBonus;

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "Your deposited card progress:";
	mes "";

	.@found = 0;

	for (.@i = 0; .@i < getarraysize(.CardID); .@i++) {
		.@amount = 0;
		query_sql("SELECT `amount` FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.CardID[.@i], .@amount);

		if (.@amount > 0) {
			.@found++;

			if (.Tier[.@i] == 2)
				mes "^FF0000[Rare]^000000 ^3366CC"+getitemname(.CardID[.@i])+"^000000 ^00AA99["+.@amount+"/"+.Required[.@i]+"]^000000";
			else
				mes "^3366CC[Normal]^000000 ^3366CC"+getitemname(.CardID[.@i])+"^000000 ^00AA99["+.@amount+"/"+.Required[.@i]+"]^000000";
		}
	}

	if (.@found <= 0)
		mes "^999999No deposited cards found.^000000";

	next;

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "Tier Status:";
	mes "^3366CCNormal MVP Tier:^000000 " + (#CardDep_NormalComplete ? "^00AA00Complete^000000" : "^FF0000Incomplete^000000");
	mes "^FF0000Rare MVP Tier:^000000 " + (#CardDep_RareComplete ? "^00AA00Complete^000000" : "^FF0000Incomplete^000000");
	return;

S_DepositMenu:
	mes "[^3366CCCard Deposit Agent^000000]";
	mes "Which card do you want to deposit?";
	next;

	.@menu$ = "";

	for (.@i = 0; .@i < getarraysize(.CardID); .@i++) {
		.@amount = 0;
		query_sql("SELECT `amount` FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.CardID[.@i], .@amount);

		if (.Tier[.@i] == 2)
			.@menu$ += "^FF0000[Rare]^000000 "+getitemname(.CardID[.@i])+" ^00AA99["+.@amount+"/"+.Required[.@i]+"]^000000:";
		else
			.@menu$ += "^3366CC[Normal]^000000 "+getitemname(.CardID[.@i])+" ^00AA99["+.@amount+"/"+.Required[.@i]+"]^000000:";
	}

	.@select = select(.@menu$) - 1;

	.@card_id = .CardID[.@select];
	.@required = .Required[.@select];
	.@tier = .Tier[.@select];

	.@current = 0;
	query_sql("SELECT `amount` FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.@card_id, .@current);

	.@have = countitem(.@card_id);

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "Selected Card:";
	mes "^3366CC"+getitemname(.@card_id)+"^000000";
	mes "";
	mes "Current Deposit: ^00AA99"+.@current+"/"+.@required+"^000000";
	mes "Card in Inventory: ^00AA00"+.@have+"^000000";
	mes "";
	mes "How many do you want to deposit?";
	next;

	if (.@have <= 0) {
		mes "[^3366CCCard Deposit Agent^000000]";
		mes "You don't have this card.";
		close;
	}

	input .@deposit_amount;

	if (.@deposit_amount <= 0) {
		mes "[^3366CCCard Deposit Agent^000000]";
		mes "Invalid amount.";
		close;
	}

	if (.@deposit_amount > .@have) {
		mes "[^3366CCCard Deposit Agent^000000]";
		mes "You don't have enough cards.";
		close;
	}

	delitem .@card_id, .@deposit_amount;

	query_sql("INSERT INTO `card_deposits` (`account_id`, `card_id`, `amount`) VALUES ("+getcharid(3)+", "+.@card_id+", "+.@deposit_amount+") ON DUPLICATE KEY UPDATE `amount` = `amount` + "+.@deposit_amount);

	callsub S_ApplyBonus;

	.@new_amount = 0;
	query_sql("SELECT `amount` FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.@card_id, .@new_amount);

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "You have deposited ^00AA00"+.@deposit_amount+"x^000000";
	mes "^3366CC"+getitemname(.@card_id)+"^000000.";
	mes "";
	mes "Current Progress: ^00AA99"+.@new_amount+"/"+.@required+"^000000";
	next;

	if (callsub(S_IsTierComplete, .@tier))
		callsub S_TryRewardClaim, .@tier;

	close;

S_WithdrawMenu:
	mes "[^3366CCCard Deposit Agent^000000]";
	mes "^FF0000Important:^000000";
	mes "If the tier becomes incomplete, the bonus stat will be removed.";
	next;

	.@menu$ = "";
	.@count = 0;

	for (.@i = 0; .@i < getarraysize(.CardID); .@i++) {
		.@amount = 0;
		query_sql("SELECT `amount` FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.CardID[.@i], .@amount);

		if (.@amount > 0) {
			.@Index[.@count] = .@i;

			if (.Tier[.@i] == 2)
				.@menu$ += "^FF0000[Rare]^000000 "+getitemname(.CardID[.@i])+" ^00AA99["+.@amount+"/"+.Required[.@i]+"]^000000:";
			else
				.@menu$ += "^3366CC[Normal]^000000 "+getitemname(.CardID[.@i])+" ^00AA99["+.@amount+"/"+.Required[.@i]+"]^000000:";

			.@count++;
		}
	}

	if (.@count <= 0) {
		mes "[^3366CCCard Deposit Agent^000000]";
		mes "You don't have any deposited cards.";
		close;
	}

	.@select = select(.@menu$) - 1;
	.@real_index = .@Index[.@select];

	.@card_id = .CardID[.@real_index];
	.@tier = .Tier[.@real_index];

	.@current = 0;
	query_sql("SELECT `amount` FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.@card_id, .@current);

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "You have ^00AA00"+.@current+"x^000000";
	mes "^3366CC"+getitemname(.@card_id)+"^000000 deposited.";
	mes "";
	mes "How many do you want to withdraw?";
	next;

	input .@withdraw_amount;

	if (.@withdraw_amount <= 0 || .@withdraw_amount > .@current) {
		mes "[^3366CCCard Deposit Agent^000000]";
		mes "Invalid amount.";
		close;
	}

	if (!checkweight(.@card_id, .@withdraw_amount)) {
		mes "[^3366CCCard Deposit Agent^000000]";
		mes "Your inventory is full or overweight.";
		close;
	}

	query_sql("UPDATE `card_deposits` SET `amount` = `amount` - "+.@withdraw_amount+" WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.@card_id);
	query_sql("DELETE FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.@card_id+" AND `amount` <= 0");

	getitem .@card_id, .@withdraw_amount;

	callsub S_ApplyBonus;

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "You have withdrawn ^FF0000"+.@withdraw_amount+"x^000000";
	mes "^3366CC"+getitemname(.@card_id)+"^000000.";

	if ((.@tier == 1 && !#CardDep_NormalComplete) || (.@tier == 2 && !#CardDep_RareComplete)) {
		mes "";
		mes "^FF0000This tier is now incomplete.^000000";
		mes "^FF0000The tier bonus stat has been removed.^000000";
	}

	close;

S_ShowTier:
	.@tier = getarg(0);

	for (.@i = 0; .@i < getarraysize(.CardID); .@i++) {
		if (.Tier[.@i] == .@tier) {
			.@amount = 0;
			query_sql("SELECT `amount` FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.CardID[.@i], .@amount);

			if (.@tier == 2)
				mes "^FF0000[Rare]^000000 ^3366CC"+getitemname(.CardID[.@i])+"^000000 ^00AA99["+.@amount+"/"+.Required[.@i]+"]^000000";
			else
				mes "^3366CC[Normal]^000000 ^3366CC"+getitemname(.CardID[.@i])+"^000000 ^00AA99["+.@amount+"/"+.Required[.@i]+"]^000000";
		}
	}

	mes "";
	if (.@tier == 1)
		mes "Status: " + (#CardDep_NormalComplete ? "^00AA00Complete^000000" : "^FF0000Incomplete^000000");
	else
		mes "Status: " + (#CardDep_RareComplete ? "^00AA00Complete^000000" : "^FF0000Incomplete^000000");

	return;

S_IsTierComplete:
	.@tier = getarg(0);

	for (.@i = 0; .@i < getarraysize(.CardID); .@i++) {
		if (.Tier[.@i] == .@tier) {
			.@amount = 0;
			query_sql("SELECT `amount` FROM `card_deposits` WHERE `account_id` = "+getcharid(3)+" AND `card_id` = "+.CardID[.@i], .@amount);

			if (.@amount < .Required[.@i])
				return 0;
		}
	}
	return 1;

S_ApplyBonus:
	#CardDep_NormalComplete = callsub(S_IsTierComplete, 1);
	#CardDep_RareComplete = callsub(S_IsTierComplete, 2);
	recalculatestat;
	return;

S_TryRewardClaim:
	.@tier = getarg(0);

	.@account_id = getcharid(3);
	.@ip$ = getcharip();
	.@unique_id$ = get_unique_id();

	.@claimed = 0;
	query_sql("SELECT COUNT(*) FROM `card_deposit_claim` WHERE `tier` = "+.@tier+" AND (`account_id` = "+.@account_id+" OR `ip` = '"+escape_sql(.@ip$)+"' OR `unique_id` = '"+escape_sql(.@unique_id$)+"')", .@claimed);

	if (.@claimed > 0) {
		mes "[^3366CCCard Deposit Agent^000000]";
		mes "^00AA00Congratulations!^000000";
		mes "You have completed this tier.";
		mes "";
		mes "^FF0000Reward item cannot be received again.^000000";
		mes "^00AA00Bonus stat is now active while this tier remains complete.^000000";
		return;
	}

	if (.@tier == 1) {
		if (!checkweight(33036,1) || !checkweight(7773,50)) {
			mes "[^3366CCCard Deposit Agent^000000]";
			mes "Your inventory is full or overweight.";
			mes "Your reward has not been claimed yet.";
			return;
		}
	}
	else {
		if (!checkweight(33037,1) || !checkweight(33048,10) || !checkweight(51002,1) || !checkweight(7773,100)) {
			mes "[^3366CCCard Deposit Agent^000000]";
			mes "Your inventory is full or overweight.";
			mes "Your reward has not been claimed yet.";
			return;
		}
	}

	mes "[^3366CCCard Deposit Agent^000000]";
	mes "^00AA00Congratulations!^000000";
	mes "You have completed the " + ((.@tier == 1) ? "Normal MVP Tier." : "Rare MVP Tier.");
	close2;

	progressbar "0x00FF00",4;

	if (.@tier == 1) {
		getitem 33036,1;
		getitem 7773,50;
	}
	else {
		getitem 33037,1;
		getitem 33048,10;
		getitem 51002,1;
		getitem 7773,100;
	}

	query_sql("INSERT INTO `card_deposit_claim` (`account_id`, `ip`, `unique_id`, `tier`, `claimed_at`) VALUES ("+.@account_id+", '"+escape_sql(.@ip$)+"', '"+escape_sql(.@unique_id$)+"', "+.@tier+", NOW())");

	callsub S_ApplyBonus;

	message strcharinfo(0), "Card Deposit bonus stat is now active.";
	dispbottom "Card Deposit bonus stat is now active.";
	end;

OnInit:
	setarray .NormalCards[0],
		4142,4144,4147,4134,4145,4121,4168,4305,4302,4430,4419,4324,4318,4441,4330,4376,4352;

	setarray .RareCards[0],
		4399,4403,4359,4367,4365,4357,4361,4363;

	.@index = 0;

	for (.@i = 0; .@i < getarraysize(.NormalCards); .@i++) {
		.CardID[.@index] = .NormalCards[.@i];
		.Required[.@index] = 100;
		.Tier[.@index] = 1;
		.@index++;
	}

	for (.@i = 0; .@i < getarraysize(.RareCards); .@i++) {
		.CardID[.@index] = .RareCards[.@i];
		.Required[.@index] = 10;
		.Tier[.@index] = 2;
		.@index++;
	}

	end;
}
