// ============================================================================
// FUNCTION: SAVE PLAYER LOOK
// ============================================================================
function	script	F_SavePvpLook	{
	if ( !getcharid(0) ) return;

	query_sql "REPLACE INTO `pvpladder_looks` "
		+"(`char_id`,`name`,`class`,`hair`,`hair_color`,`cloth_color`,`weapon`,`shield`,`head_top`,`head_mid`,`head_bottom`,`updated_at`) VALUES ("
		+ getcharid(0) +","
		+"'"+ escape_sql(strcharinfo(0)) +"',"
		+ Class +","
		+ getlook(LOOK_HAIR) +","
		+ getlook(LOOK_HAIR_COLOR) +","
		+ getlook(LOOK_CLOTHES_COLOR) +","
		+ getlook(LOOK_WEAPON) +","
		+ getlook(LOOK_SHIELD) +","
		+ getlook(LOOK_HEAD_TOP) +","
		+ getlook(LOOK_HEAD_MIDDLE) +","
		+ getlook(LOOK_HEAD_BOTTOM) +","
		+"NOW())";
	return;
}

// ============================================================================
// FUNCTION: RESET STATUE LOOK
// ============================================================================
function	script	F_ResetPvpStatue	{
	.@npc$ = getarg(0);
	.@gid = getnpcid(0,.@npc$);
	if ( !.@gid ) return;

	setunitdata .@gid, UNPC_CLASS, Job_Novice;
	setunitdata .@gid, UNPC_HAIRSTYLE, 0;
	setunitdata .@gid, UNPC_HAIRCOLOR, 0;
	setunitdata .@gid, UNPC_CLOTHCOLOR, 0;
	setunitdata .@gid, UNPC_WEAPON, 0;
	setunitdata .@gid, UNPC_SHIELD, 0;
	setunitdata .@gid, UNPC_HEADTOP, 0;
	setunitdata .@gid, UNPC_HEADMIDDLE, 0;
	setunitdata .@gid, UNPC_HEADBOTTOM, 0;
	return;
}

// ============================================================================
// FUNCTION: APPLY STATUE LOOK
// ============================================================================
function	script	F_ApplyPvpStatue	{
	.@npc$ = getarg(0);
	.@class = getarg(1);
	.@hair = getarg(2);
	.@hair_color = getarg(3);
	.@cloth_color = getarg(4);
	.@weapon = getarg(5);
	.@shield = getarg(6);
	.@head_top = getarg(7);
	.@head_mid = getarg(8);
	.@head_bottom = getarg(9);

	.@gid = getnpcid(0,.@npc$);
	if ( !.@gid ) return;

	setunitdata .@gid, UNPC_CLASS, .@class;
	setunitdata .@gid, UNPC_HAIRSTYLE, .@hair;
	setunitdata .@gid, UNPC_HAIRCOLOR, .@hair_color;
	setunitdata .@gid, UNPC_CLOTHCOLOR, .@cloth_color;
	setunitdata .@gid, UNPC_WEAPON, .@weapon;
	setunitdata .@gid, UNPC_SHIELD, .@shield;
	setunitdata .@gid, UNPC_HEADTOP, .@head_top;
	setunitdata .@gid, UNPC_HEADMIDDLE, .@head_mid;
	setunitdata .@gid, UNPC_HEADBOTTOM, .@head_bottom;
	return;
}
