//==============================================================
// [1] SQL PERSISTENT DEFENDER SAVE
//==============================================================
//
// PURPOSE:
// Saves the defending guild even after server restart.
//
//==============================================================

// SQL TABLE

CREATE TABLE IF NOT EXISTS `guild_throne_data` (
	`id` int NOT NULL AUTO_INCREMENT,
	`defender_guild_id` int NOT NULL DEFAULT 0,
	`last_winner` varchar(50) NOT NULL DEFAULT '',
	`win_count` int NOT NULL DEFAULT 0,
	`last_event` datetime NULL,
	PRIMARY KEY (`id`)
);

//==============================================================
// LOAD DEFENDER ON SERVER START
//==============================================================

OnInit:

	query_sql(
		"SELECT defender_guild_id FROM guild_throne_data LIMIT 1",
		.DefenderGuild
	);

	if (.DefenderGuild > 0)
		debugmes "[Guild Throne War] Loaded Defender Guild ID: "+.DefenderGuild;

	end;

//==============================================================
// SAVE DEFENDER AFTER WIN
//==============================================================

function	script	F_GTW_SaveDefender	{

	.@gid = getarg(0);

	query_sql(
		"DELETE FROM guild_throne_data"
	);

	query_sql(
		"INSERT INTO guild_throne_data(defender_guild_id,last_winner,last_event) VALUES ("+
		.@gid+",'"+escape_sql(getguildname(.@gid))+"',NOW())"
	);

	return;
}



//==============================================================
// [2] RANKING SYSTEM
//==============================================================
//
// PURPOSE:
// Stores win rankings.
//
//==============================================================

CREATE TABLE IF NOT EXISTS `guild_throne_rankings` (
	`guild_id` int NOT NULL,
	`guild_name` varchar(50) NOT NULL,
	`wins` int NOT NULL DEFAULT 0,
	PRIMARY KEY (`guild_id`)
);

//==============================================================
// ADD WIN
//==============================================================

function	script	F_GTW_AddWin	{

	.@gid = getarg(0);

	query_sql(
		"INSERT INTO guild_throne_rankings(guild_id,guild_name,wins) "+
		"VALUES ("+.@gid+",'"+escape_sql(getguildname(.@gid))+"',1) "+
		"ON DUPLICATE KEY UPDATE wins = wins + 1"
	);

	return;
}

		.@menu$ += getguildname(.Guilds[.@i])+":";

	close;

//==============================================================
// [5] ANTI DUAL CLIENT
//==============================================================
//
// PURPOSE:
// Prevent same IP joining multiple accounts.
//
//==============================================================

function	script	F_GTW_CheckDualIP	{

	.@gid = getarg(0);

	getguildmember .@gid,1;

	for (.@i = 0; .@i < $@guildmembercount; .@i++) {

		if (!isloggedin($@guildmemberaid[.@i],$@guildmembercid[.@i]))
			continue;

		attachrid $@guildmemberaid[.@i];

		.@ip$ = getcharip();

		for (.@k = .@i + 1; .@k < $@guildmembercount; .@k++) {

			attachrid $@guildmemberaid[.@k];

			if (.@ip$ == getcharip())
				return 1;
		}
	}

	return 0;
}

//==============================================================
// USAGE
//==============================================================

if (callfunc("F_GTW_CheckDualIP",.@gid)) {
	mes "Dual client detected.";
	close;
}



//==============================================================
// [6] MAX MEMBER LIMIT
//==============================================================
//
// PURPOSE:
// Limit participants inside event.
//
//==============================================================

.MaxInside = 20;

//==============================================================

function	script	F_GTW_WarpLimitedGuild	{

	.@map$ = getarg(0);
	.@x = getarg(1);
	.@y = getarg(2);
	.@gid = getarg(3);

	getguildmember .@gid,1;

	.@count = 0;

	for (.@i = 0; .@i < $@guildmembercount; .@i++) {

		if (.@count >= .MaxInside)
			break;

		if (!isloggedin($@guildmemberaid[.@i],$@guildmembercid[.@i]))
			continue;

		attachrid $@guildmemberaid[.@i];

		warp .@map$,.@x,.@y;

		.@count++;
	}

	return;
}

.QueueCount++;

//==============================================================
// NEXT GUILD
//==============================================================

function	script	F_GTW_GetNextQueue	{

	if (.QueueCount <= 0)
		return 0;

	.@gid = .QueueGuilds[0];

	deletearray .QueueGuilds[0],1;

	.QueueCount--;

	return .@gid;
}
