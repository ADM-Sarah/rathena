//============================================================
// Premium Gacha Machine
// Required Item: 3 Nostalgic Coin per Roll
// Fixed Real Percentage Version
//============================================================

valmorra,206,213,4	script	Gacha Machine	563,{

	.@name$ = strcharinfo(0);
	.@ticket = .itm_p;

	setdialogsize(.DialogW,.DialogH);
	setdialogalign(DIALOG_ALIGN_CENTER);

	mes "<b>^000000[ ^FF6600Premium Gacha Machine^000000 ]</b>";
	mes "^000000Hello Adventurer!^000000";
	mes "^000000Welcome to the ^FF6600Premium Gacha Machine^000000.";
	mes "^000000Use ^00CC66Nostalgic Coin^000000 to roll for random prizes.";
	mes "^777777--------------------------------------------------^000000";
	mes "^000000Required Item:^000000 ^00CC66^i["+.@ticket+"] "+getitemname(.@ticket)+"^000000 x"+.itm_a+" per roll";
	next;

	mes "<b>^000000[ ^FF6600Reward List^000000 ]</b>";
	mes "^33AAFF[5%]^000000 ^i[7619] Enriched Elunium";
	mes "^33AAFF[5%]^000000 ^i[7620] Enriched Oridecon";
	mes "^33AAFF[15%]^000000 ^i[55557] Scroll Buff";
	mes "^FFAA33[1%]^000000 ^i[51098] Fusion Valkyrie Helm";
	mes "^FFAA33[1%]^000000 ^i[51097] Fusion LHZ Aura";
	mes "^FFAA33[1%]^000000 ^i[51099] Fusion Wings";
	mes "^FFAA33[1%]^000000 ^i[51028] Baryon Ear";
	mes "^FFAA33[1%]^000000 ^i[51027] Baryon Aura";
	mes "^FFAA33[1%]^000000 ^i[51029] Baryon Tail";
	mes "^66CC66[10%]^000000 ^i[51004] Nostalgic Event Ticket";
	mes "^6699FF[10%]^000000 ^i[51005] Nostalgic MVP Ticket";
	mes "^FF0000[1%]^000000 ^i[33034] VIP Ticket 1 Day";
	mes "^FF0000[10%]^000000 ^i[33021] Yggdrasil Berry Box";
	mes "^FF0000[10%]^000000 ^i[12534] Yggdrasil Seed Box";
	mes "^777777[28%]^000000 ^i[51001] Nostalgic Pot Box";
	next;

	switch(select(
		"^3CB371» 1 Roll ^777777(3 Nostalgic Coin)^000000:"
		+ "^3CB371» 10 Rolls ^777777(30 Nostalgic Coin)^000000:"
		+ "^3CB371» 20 Rolls ^777777(60 Nostalgic Coin)^000000:"
		+ "^FF0000» Cancel^000000"
	)) {
	case 1:
		.@count = 1;
		break;
	case 2:
		.@count = 10;
		break;
	case 3:
		.@count = 20;
		break;
	default:
		close;
	}

	.@need = .itm_a * .@count;

	if (countitem(.@ticket) < .@need) {
		mes "<b>^000000[ ^FF6600Premium Gacha Machine^000000 ]</b>";
		mes "^FF0000You do not have enough ^00CC66"+getitemname(.@ticket)+"^FF0000.^000000";
		mes "^000000Required:^000000 ^FF3300"+.@need+"^000000";
		mes "^000000Your Coin:^000000 ^00CC66"+countitem(.@ticket)+"^000000";
		close;
	}

	if (!checkweight(501,.@count)) {
		mes "<b>^000000[ ^FF6600Premium Gacha Machine^000000 ]</b>";
		mes "^FF0000You are carrying too much weight.^000000";
		mes "^000000Please free some inventory space first.^000000";
		close;
	}

	mes "<b>^000000[ ^FF6600Premium Gacha Machine^000000 ]</b>";
	mes "^000000You are about to spend ^FF3300"+.@need+"^000000 ^00CC66Nostalgic Coin(s)^000000.";
	mes "^000000Do you want to continue?^000000";
	next;

	if (select("^3CB371» Yes, roll now^000000:^FF0000» No, cancel^000000") == 2)
		close;

	delitem .@ticket,.@need;

	specialeffect2 EF_BLESSING;

	mes "<b>^000000[ ^FF6600Premium Gacha Machine^000000 ]</b>";
	mes "^000000Rolling the machine...^000000";
	progressbar "0x11CC99", .loading;
	specialeffect2 EF_REMOVETRAP;
	next;

	.@summary$ = "";
	.@rare_count = 0;

	for (.@i = 0; .@i < .@count; .@i++) {

		.@r = rand(10000);
		.@rare = 0;

		if (.@r < 500) { // 5%
			.@itemid = 7619;
			.@amount = 1;
		}
		else if (.@r < 1000) { // 5%
			.@itemid = 7620;
			.@amount = 1;
		}
		else if (.@r < 2500) { // 15%
			.@itemid = 55557;
			.@amount = 1;
		}
		else if (.@r < 2600) { // 1%
			.@itemid = 51098;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 2700) { // 1%
			.@itemid = 51097;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 2800) { // 1%
			.@itemid = 51099;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 2900) { // 1%
			.@itemid = 51028;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 3000) { // 1%
			.@itemid = 51027;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 3100) { // 1%
			.@itemid = 51029;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 4100) { // 10%
			.@itemid = 51004;
			.@amount = 1;
		}
		else if (.@r < 5100) { // 10%
			.@itemid = 51005;
			.@amount = 1;
		}
		else if (.@r < 5200) { // 1%
			.@itemid = 33034;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 6200) { // 10%
			.@itemid = 33021;
			.@amount = 1;
		}
		else if (.@r < 7200) { // 10%
			.@itemid = 12534;
			.@amount = 1;
		}
		else { // 28%
			.@itemid = 51001;
			.@amount = 1;
		}

		getitem .@itemid,.@amount;

		if (.@summary$ == "")
			.@summary$ = "^0055FF- ^000000^i["+.@itemid+"] "+getitemname(.@itemid)+" x"+.@amount;
		else
			.@summary$ += "\n^0055FF- ^000000^i["+.@itemid+"] "+getitemname(.@itemid)+" x"+.@amount;

		dispbottom "[Gacha] You obtained: "+getitemname(.@itemid)+" x"+.@amount;

		if (.@rare) {
			.@rare_count++;
			specialeffect2 EF_LEVELUP;
			announce "[Gacha] "+.@name$+" obtained "+getitemname(.@itemid)+" from Premium Gacha Machine!", bc_map;
		}
	}

	mes "<b>^000000[ ^FF6600Premium Gacha Machine^000000 ]</b>";
	mes "^000000Your roll has been completed.^000000";
	mes "";
	mes "^0000FFYou received:^000000";
	mes .@summary$;
	next;

	mes "<b>^000000[ ^FF6600Premium Gacha Machine^000000 ]</b>";
	mes "^000000Total Rolls:^000000 ^FF3300"+.@count+"^000000";
	mes "^000000Coin Used:^000000 ^00CC66"+.@need+" Nostalgic Coin(s)^000000";
	mes "^000000Rare Rewards:^000000 ^FF3300"+.@rare_count+"^000000";
	mes "";
	mes "^000000Thank you for using the ^FF6600Premium Gacha Machine^000000.^000000";
	close;

OnInit:
	.n$ = "[ Gacha Machine ]";
	.DialogW = 360;
	.DialogH = 300;
	.loading = 1;

	// Required Item
	.itm_p = 33048; // Nostalgic Coin
	.itm_a = 3;     // 3 Nostalgic Coin per roll

	end;
}
