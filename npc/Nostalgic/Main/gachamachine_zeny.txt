//============================================================
// Zeny Gacha Machine
// Location: valmorra 202,213
// Cost: 50,000,000 Zeny per Roll
//============================================================

valmorra,202,213,4	script	Zeny Gacha Machine	563,{

	.@name$ = strcharinfo(0);

	setdialogsize(.DialogW,.DialogH);
	setdialogalign(DIALOG_ALIGN_CENTER);

	mes "<b>^000000[ ^FF6600Zeny Gacha Machine^000000 ]</b>";
	mes "^000000Hello Adventurer!^000000";
	mes "^000000Welcome to the ^FF6600Zeny Gacha Machine^000000.";
	mes "^000000Use ^FFCC00Zeny^000000 to roll for random prizes.";
	mes "^777777--------------------------------------------------^000000";
	mes "^000000Cost:^000000 ^FF3300"+callfunc("F_InsertComma", .zeny_per_roll)+" Zeny^000000 per roll";
	next;

	mes "<b>^000000[ ^FF6600Reward List^000000 ]</b>";
	mes "^33AAFF[1%]^000000 ^i[7619] Enriched Elunium";
	mes "^33AAFF[1%]^000000 ^i[7620] Enriched Oridecon";
	mes "^33AAFF[15%]^000000 ^i[55557] Scroll Buff";
	mes "^66CC66[1%]^000000 ^i[51004] Nostalgic Event Ticket";
	mes "^6699FF[1%]^000000 ^i[51005] Nostalgic MVP Ticket";
	mes "^FF0000[1%]^000000 ^i[33034] VIP Ticket 1 Day";
	mes "^FF0000[10%]^000000 ^i[33021] Yggdrasil Berry Box";
	mes "^FF0000[10%]^000000 ^i[12534] Yggdrasil Seed Box";
	mes "^777777[30%]^000000 ^i[51001] Nostalgic Pot Box";
	mes "^777777[30%]^000000 ^i[33017] Credit";
	next;

	switch(select(
		"^3CB371» 1 Roll ^777777(100M Zeny)^000000:"
		+ "^3CB371» 10 Rolls ^777777(1B Zeny)^000000:"
		+ "^3CB371» 20 Rolls ^777777(2B Zeny)^000000:"
		+ "^FF0000» Cancel^000000"
	)) {
	case 1:
		.@count = 1;
		break;
	case 2:
		.@count = 10;
		break;
	case 3:
		.@count = 20;
		break;
	default:
		close;
	}

	.@need = .zeny_per_roll * .@count;

	if (Zeny < .@need) {
		mes "<b>^000000[ ^FF6600Zeny Gacha Machine^000000 ]</b>";
		mes "^FF0000You do not have enough Zeny.^000000";
		mes "^000000Required:^000000 ^FF3300"+callfunc("F_InsertComma", .@need)+" Zeny^000000";
		mes "^000000Your Zeny:^000000 ^00CC66"+callfunc("F_InsertComma", Zeny)+" Zeny^000000";
		close;
	}

	if (!checkweight(501,.@count)) {
		mes "<b>^000000[ ^FF6600Zeny Gacha Machine^000000 ]</b>";
		mes "^FF0000You are carrying too much weight.^000000";
		mes "^000000Please free some inventory space first.^000000";
		close;
	}

	mes "<b>^000000[ ^FF6600Zeny Gacha Machine^000000 ]</b>";
	mes "^000000You are about to spend:^000000";
	mes "^FF3300"+callfunc("F_InsertComma", .@need)+" Zeny^000000";
	mes "";
	mes "^000000Do you want to continue?^000000";
	next;

	if (select("^3CB371» Yes, roll now^000000:^FF0000» No, cancel^000000") == 2)
		close;

	Zeny -= .@need;

	specialeffect2 EF_BLESSING;

	mes "<b>^000000[ ^FF6600Zeny Gacha Machine^000000 ]</b>";
	mes "^000000Rolling the machine...^000000";
	progressbar "0x11CC99", .loading;
	specialeffect2 EF_REMOVETRAP;
	next;

	.@summary$ = "";
	.@rare_count = 0;

	for (.@i = 0; .@i < .@count; .@i++) {

		.@r = rand(10000);
		.@rare = 0;

		if (.@r < 100) { // 1%
			.@itemid = 7619;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 200) { // 1%
			.@itemid = 7620;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 1700) { // 15%
			.@itemid = 55557;
			.@amount = 1;
		}
		else if (.@r < 1800) { // 1%
			.@itemid = 51004;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 1900) { // 1%
			.@itemid = 51005;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 2000) { // 1%
			.@itemid = 33034;
			.@amount = 1;
			.@rare = 1;
		}
		else if (.@r < 3000) { // 10%
			.@itemid = 33021;
			.@amount = 1;
		}
		else if (.@r < 4000) { // 10%
			.@itemid = 12534;
			.@amount = 1;
		}
		else if (.@r < 7000) { // 30%
			.@itemid = 51001;
			.@amount = 1;
		}
		else { // 30%
			.@itemid = 33017;
			.@amount = 1;
		}

		getitem .@itemid,.@amount;

		if (.@summary$ == "")
			.@summary$ = "^0055FF- ^000000^i["+.@itemid+"] "+getitemname(.@itemid)+" x"+.@amount;
		else
			.@summary$ += "\n^0055FF- ^000000^i["+.@itemid+"] "+getitemname(.@itemid)+" x"+.@amount;

		dispbottom "[Gacha] You obtained: "+getitemname(.@itemid)+" x"+.@amount;

		if (.@rare) {
			.@rare_count++;
			specialeffect2 EF_LEVELUP;
			announce "[Gacha] "+.@name$+" obtained "+getitemname(.@itemid)+" from Zeny Gacha Machine!", bc_map;
		}
	}

	mes "<b>^000000[ ^FF6600Zeny Gacha Machine^000000 ]</b>";
	mes "^000000Your roll has been completed.^000000";
	mes "";
	mes "^0000FFYou received:^000000";
	mes .@summary$;
	next;

	mes "<b>^000000[ ^FF6600Zeny Gacha Machine^000000 ]</b>";
	mes "^000000Total Rolls:^000000 ^FF3300"+.@count+"^000000";
	mes "^000000Zeny Used:^000000 ^FF3300"+callfunc("F_InsertComma", .@need)+" Zeny^000000";
	mes "^000000Rare Rewards:^000000 ^FF3300"+.@rare_count+"^000000";
	mes "";
	mes "^000000Thank you for using the ^FF6600Zeny Gacha Machine^000000.^000000";
	close;

OnInit:
	.DialogW = 360;
	.DialogH = 300;
	.loading = 1;

	.zeny_per_roll = 100000000; // 50M Zeny per roll

	end;
}
