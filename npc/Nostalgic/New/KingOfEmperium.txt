			setdialogsize(440,260);
			setdialogalign(DIALOG_ALIGN_CENTER);
			end;

	end;

//------------------------------------
// Timers
//------------------------------------
OnStart:
OnMinute00:
	.Size = getarraysize($KOE_CONTROL);
	for (.@i = 0; .@i < .Size; .@i += 3)
		if (gettime(DT_DAYOFWEEK) == $KOE_CONTROL[.@i] - 1 && gettime(DT_HOUR) == $KOE_CONTROL[.@i + 1] - 1)
			donpcevent "KoE#" + .Maps$[$KOE_CONTROL[.@i + 2] - 1] + "::OnStartKoE";

		$@koeactive = 0;
		for (.@i = 0; .@i < getarraysize(.Maps$); .@i++)
			if (getd("$@koe_" + .Maps$[.@i] + "_start")) {
				$@koeactive = 1;
				break;
			}
		if (!$@koeactive) break;
		specialeffect EF_HOMUNCASTING;
		sleep 500;
	}
	end;

OnEnd:
OnMinute50:
	.Size = getarraysize($KOE_CONTROL);
	for (.@i = 0; .@i < .Size; .@i += 3)
		if (gettime(DT_DAYOFWEEK) == $KOE_CONTROL[.@i] - 1 && gettime(DT_HOUR) == $KOE_CONTROL[.@i + 1] - 1)
			donpcevent "KoE#" + .Maps$[$KOE_CONTROL[.@i + 2] - 1] + "::OnEndKoE";
	end;

//------------------------------------
// Schedule viewer
//------------------------------------
OnSchedule:
	freeloop(1);
	for (.@j = 0; .@j < 7; .@j++) {
		mes "<b>^000000[ ^FF7F00King of Emperium^000000 ]</b>";
		mes "^000000King of Emperium Schedule^000000";
		mes "^1E90FF" + .Days$[.@j] + "^000000:";

		for (.@i = 0; .@i < .Size; .@i += 3)
			if ($KOE_CONTROL[.@i] - 1 == .@j)
				mes "^FF7F00» ^000000^1E90FF" + .MapsName$[$KOE_CONTROL[.@i + 2] - 1] + "^000000 [ " + Add_Zero($KOE_CONTROL[.@i + 1] - 1) + " - " + Add_Zero2($KOE_CONTROL[.@i + 1] - 1) + " ]";

		setdialogsize(440,260);
		setdialogalign(DIALOG_ALIGN_CENTER);
		next;
	}
	freeloop(0);
	return;

function Add_Zero {
	return ((getarg(0) < 10) ? "0" : "") + getarg(0) + (getarg(1, 0) ? "." : ":") + "00";
}

function Add_Zero2 {
	return ((getarg(0) < 10) ? "0" : "") + getarg(0) + (getarg(1, 0) ? "." : ":") + "50";
}
}

	deletearray getd(".koe_" + strnpcinfo(2) + "_def[0]"), .@size;
	deletearray getd(".koe_" + strnpcinfo(2) + "_time[0]"), .@size;
	setd ".koe_" + strnpcinfo(2) + "_defguild", 0;
	setd ".koe_" + strnpcinfo(2) + "_deftime", 0;

	announce "The King of Emperium Hill in " + callsub(OnKoeMap) + " has started! Prove who is the best guild!", 0;
	donpcevent "::OnRevKoE";
	maprespawnguildid strnpcinfo(2), getd("$koe_" + strnpcinfo(2)), 6;
	callsub OnSetKoE;
	end;

OnWhisperGlobal:
	if (getgmlevel() < 99) end;

OnEndKoE:
	if (!getd("$@koe_" + strnpcinfo(2) + "_start")) end;
	announce "The King of Emperium Hill in " + callsub(OnKoeMap) + " has ended!", 0;

	setcell strnpcinfo(2), getd(".walkable_" + strnpcinfo(2) + "[0]"), getd(".walkable_" + strnpcinfo(2) + "[1]"), getd(".walkable_" + strnpcinfo(2) + "[2]"), getd(".walkable_" + strnpcinfo(2) + "[3]"), cell_walkable, 1;
	setd "$@koe_" + strnpcinfo(2) + "_start", 0;

	callsub OnDefTime;
	for (.@i = 0; .@i < getarraysize(getd(".koe_" + strnpcinfo(2) + "_def")); .@i++) {
		if (getd(".koe_" + strnpcinfo(2) + "_deftime") < getd(".koe_" + strnpcinfo(2) + "_time[" + .@i + "]")) {
			setd ".koe_" + strnpcinfo(2) + "_defguild", getd(".koe_" + strnpcinfo(2) + "_def[" + .@i + "]");
			setd ".koe_" + strnpcinfo(2) + "_deftime", getd(".koe_" + strnpcinfo(2) + "_time[" + .@i + "]");
		}
	}

	if (getd(".koe_" + strnpcinfo(2) + "_defguild")) {
		query_sql("INSERT INTO `kingofemp` (`type`, `name`, `val1`, `val2`) VALUES ('1', '" + getguildname(getd(".koe_" + strnpcinfo(2) + "_defguild")) + "', '" + getd(".koe_" + strnpcinfo(2) + "_defguild") + "', '" + getd(".koe_" + strnpcinfo(2) + "_deftime") + "');");
		query_sql("INSERT INTO `kingofemp` (`type`, `name`, `val1`) VALUES ('2', '" + getguildname(getd(".koe_" + strnpcinfo(2) + "_defguild")) + "', '" + getd("$koe_" + strnpcinfo(2)) + "');");

		announce "The guild " + getguildname(getd(".koe_" + strnpcinfo(2) + "_defguild")) + " defended the Emperium in " + callsub(OnKoeMap) + " for the longest time. " + callsub(OnConvertTime, getd(".koe_" + strnpcinfo(2) + "_deftime")), 0;

		if ((.Options & 1)) {
			callsub OnRewardDefender, getd(".koe_" + strnpcinfo(2) + "_defguild");
			callsub OnRewardHolder, getd("$koe_" + strnpcinfo(2));
		}
	}

	callsub OnRewardLosers;
	killmonsterall strnpcinfo(2);
	sleep 10000;
	mapwarp strnpcinfo(2), "prontera", 155, 184;
	end;

OnConvertTime:
	.@Time = getarg(0);
	.@Hours = .@Time / 3600000;
	.@Time = .@Time - (.@Hours * 3600000);
	.@Minutes = .@Time / 60000;
	.@Time = .@Time - (.@Minutes * 60000);
	.@Seconds = .@Time / 1000;
	.@Time = .@Time - (.@Seconds * 1000);
	.@timestr$ = (.@Hours ? .@Hours + "h " : "") + (.@Minutes ? .@Minutes + "m " : "") + (.@Seconds ? .@Seconds + "s " : "") + (.@Time ? .@Time + "ms " : "");
	return .@timestr$;

OnEmpDead:
	query_sql("INSERT INTO `kingofemp` (`name`, `val1`) VALUES ('" + escape_sql(strcharinfo(0)) + "', '" + getcharid(0) + "');");

	if (getd("$koe_" + strnpcinfo(2))) callsub OnDefTime;
	initnpctimer strnpcinfo(0);
	SetCastleData strnpcinfo(2), CD_GUILD_ID, getcharid(2);
	setd "$koe_" + strnpcinfo(2), getcharid(2);

	query_sql("REPLACE INTO `koe_runtime` (`map`,`guild_id`) VALUES ('" + strnpcinfo(2) + "','" + getcharid(2) + "')");

	for (.@i = 0; .@i < getarraysize(.maps$); .@i += 2) {
		.@gid = 0;
		query_sql("SELECT `guild_id` FROM `koe_runtime` WHERE `map` = '" + .maps$[.@i] + "'", .@gid);
		if (.@gid > 0)
			setd "$koe_" + .maps$[.@i], .@gid;
	}

	announce "The current dominant guild of King of Emperium Hill in " + callsub(OnKoeMap) + " is [ " + strcharinfo(2) + " ].", 0;
	donpcevent "::OnRevKoE";
	maprespawnguildid strnpcinfo(2), getd("$koe_" + strnpcinfo(2)), 6;
	sleep 500;

	if (getd("$@koe_" + strnpcinfo(2) + "_start")) {
		mapwarp strnpcinfo(2), strnpcinfo(2), getd(".warpbreak_" + strnpcinfo(2) + "[0]"), getd(".warpbreak_" + strnpcinfo(2) + "[1]"), 1, getd("$koe_" + strnpcinfo(2));
		callsub OnSetKoE;

		if (getd("." + strnpcinfo(2) + "_guardian")) {
			for (.@i = 0; .@i < getarraysize(getd("." + strnpcinfo(2) + "_guardian")); .@i += 3)
				guardian strnpcinfo(2), getd("." + strnpcinfo(2) + "_guardian[" + (.@i + 1) + "]"), getd("." + strnpcinfo(2) + "_guardian[" + (.@i + 2) + "]"), "Emperium Guardian", getd("." + strnpcinfo(2) + "_guardian[" + .@i + "]");
		}
	}
	end;

OnDefTime:
	.@tick = getnpctimer(0), strnpcinfo(0);
	.@a = inarray(getd(".koe_" + strnpcinfo(2) + "_def"), getd("$koe_" + strnpcinfo(2)));
	if (.@a >= 0)
		setd ".koe_" + strnpcinfo(2) + "_time[" + .@a + "]", getd(".koe_" + strnpcinfo(2) + "_time[" + .@a + "]") + .@tick;
	else {
		.@size = getarraysize(getd(".koe_" + strnpcinfo(2) + "_def"));
		setd ".koe_" + strnpcinfo(2) + "_def[" + .@size + "]", getd("$koe_" + strnpcinfo(2));
		setd ".koe_" + strnpcinfo(2) + "_time[" + .@size + "]", getd(".koe_" + strnpcinfo(2) + "_time[" + .@size + "]") + .@tick;
	}
	return;

OnSetKoE:
	killmonsterall strnpcinfo(2);
	monster strnpcinfo(2), getd("." + strnpcinfo(2) + "_emp[0]"), getd("." + strnpcinfo(2) + "_emp[1]"), "EMPERIUM", .Emperium_id, 1, strnpcinfo(0) + "::OnEmpDead";

	if (.enableBarricade) {
		setd ".EmpID_" + strnpcinfo(2), $@mobid[0];
		setunitdata getd(".EmpID_" + strnpcinfo(2)), UMOB_DMGIMMUNE, 1;
		setd ".Barricade_" + strnpcinfo(2), 0;

		for (.@i = 0; .@i < getarraysize(getd(".barx_" + strnpcinfo(2))); .@i++) {
			if (getd("$koe_" + strnpcinfo(2)))
				guardian strnpcinfo(2), getd(".barx_" + strnpcinfo(2) + "[" + .@i + "]"), getd(".bary_" + strnpcinfo(2) + "[" + .@i + "]"), "Barricade", .Barricade_id, strnpcinfo(0) + "::OnBarDead";
			else
				monster strnpcinfo(2), getd(".barx_" + strnpcinfo(2) + "[" + .@i + "]"), getd(".bary_" + strnpcinfo(2) + "[" + .@i + "]"), "Barricade", .Barricade_id, 1, strnpcinfo(0) + "::OnBarDead", 0;

			setd ".Barricade_" + strnpcinfo(2), getd(".Barricade_" + strnpcinfo(2)) + 1;
		}

		setcell strnpcinfo(2), getd(".walkable_" + strnpcinfo(2) + "[0]"), getd(".walkable_" + strnpcinfo(2) + "[1]"), getd(".walkable_" + strnpcinfo(2) + "[2]"), getd(".walkable_" + strnpcinfo(2) + "[3]"), cell_walkable, 0;
	}
	return;

OnBarDead:
	setd ".Barricade_" + strnpcinfo(2), getd(".Barricade_" + strnpcinfo(2)) - 1;
	if (getd(".Barricade_" + strnpcinfo(2)) <= 0) {
		mapannounce strnpcinfo(2), "The barricades have fallen! The Emperium is now vulnerable!", bc_map;
		setcell strnpcinfo(2), getd(".walkable_" + strnpcinfo(2) + "[0]"), getd(".walkable_" + strnpcinfo(2) + "[1]"), getd(".walkable_" + strnpcinfo(2) + "[2]"), getd(".walkable_" + strnpcinfo(2) + "[3]"), cell_walkable, 1;
		setunitdata getd(".EmpID_" + strnpcinfo(2)), UMOB_DMGIMMUNE, 0;
	}
	end;

OnGiveKoEReward:
	.@aid = getarg(0);
	.@points = getarg(1);
	.@berry = getarg(2);
	.@seed = getarg(3);
	.@type$ = getarg(4);

	if (!attachrid(.@aid))
		return;

	#KOEPOINTS += .@points;

	if (.@berry > 0)
		getitem 33021, .@berry; // Yggdrasil Berry Box

	if (.@seed > 0)
		getitem 33022, .@seed; // Yggdrasil Seed Box

	dispbottom "[KoE Reward]: You received " + .@points + " KoE Point(s), " + .@berry + " Ygg Berry Box(es), and " + .@seed + " Ygg Seed Box(es).";
	dispbottom "[KoE Reward]: Reward Type - " + .@type$ + ".";

	detachrid;
	return;

OnRewardLosers:
	freeloop(1);
	.@size = getarraysize(getd(".koe_part_n_" + strnpcinfo(2)));
	copyarray .@aid[0], getd(".koe_part_n_" + strnpcinfo(2) + "[0]"), .@size;
	copyarray .@time[0], getd(".koe_part_t_" + strnpcinfo(2) + "[0]"), .@size;

	for (.@j = 0; .@j < .@size; .@j += 1) {
		// Minimum participation: 10 active minutes.
		if (.@time[.@j] < .timepart) continue;

		if (isloggedin(.@aid[.@j]))
			callsub OnGiveKoEReward, .@aid[.@j], .part_points, .part_berry, .part_seed, "Participation";
	}
	freeloop(0);
	return;

OnRewardHolder:
	freeloop(1);
	.@gid = getarg(0);
	getguildmember .@gid, 1, .@cid;
	getguildmember .@gid, 2, .@aid;

	.@size = $@guildmembercount;

	for (.@j = 0; .@j < .@size; .@j += 1) {
		if (isloggedin(.@aid[.@j], .@cid[.@j]))
			callsub OnGiveKoEReward, .@aid[.@j], .holder_points, .holder_berry, .holder_seed, "Holder Guild";
	}
	freeloop(0);
	return;

OnRewardDefender:
	freeloop(1);
	.@gid = getarg(0);
	getguildmember .@gid, 1, .@cid;
	getguildmember .@gid, 2, .@aid;

	.@size = $@guildmembercount;

	for (.@j = 0; .@j < .@size; .@j += 1) {
		if (isloggedin(.@aid[.@j], .@cid[.@j]))
			callsub OnGiveKoEReward, .@aid[.@j], .defender_points, .defender_berry, .defender_seed, "Top Defender Guild";
	}
	freeloop(0);
	return;

OnKoeMap:
	for (.@i = 0; .@i < getarraysize(.maps$); .@i += 2)
		if (.maps$[.@i] == strnpcinfo(2))
			return .maps$[.@i + 1];
	return;

OnPCLoadMapEvent:
	if (.Participation)
	if (@lastmap$ != strnpcinfo(2) && strcharinfo(3) == strnpcinfo(2) && getd("$@koe_" + strnpcinfo(2) + "_start")) {
		set @lastmap$, strcharinfo(3);
		message strcharinfo(0), "KoE point acquisition activated.";
		addtimer 60000, strnpcinfo(0) + "::OnParticipate";
	}
	end;

OnParticipate:
	set @lastmap$, strcharinfo(3);

	if (@lastmap$ == strnpcinfo(2) && getd("$@koe_" + strnpcinfo(2) + "_start")) {
		addtimer 60000, strnpcinfo(0) + "::OnParticipate";

		// Anti-AFK: idle more than 4 minutes = no timer count and no KoE point.
		if (checkidle() >= .idle_limit) {
			dispbottom "[KoE]: You are idle for more than 4 minutes. KoE point and participation timer paused.";
			end;
		}

		#KOEPOINTS += 1;
		dispbottom "[KoE]: You received 1 KoE Point. Current KoE Points: " + #KOEPOINTS + ".";

		.@arr = inarray(getd(".koe_part_n_" + strnpcinfo(2)), getcharid(3));
		if (.@arr >= 0) {
			setd ".koe_part_t_" + strnpcinfo(2) + "[" + .@arr + "]", getd(".koe_part_t_" + strnpcinfo(2) + "[" + .@arr + "]") + 1;
			.@current_min = getd(".koe_part_t_" + strnpcinfo(2) + "[" + .@arr + "]");
		} else {
			.@idx = getarraysize(getd(".koe_part_n_" + strnpcinfo(2)));
			setd ".koe_part_n_" + strnpcinfo(2) + "[" + .@idx + "]", getcharid(3);
			setd ".koe_part_t_" + strnpcinfo(2) + "[" + .@idx + "]", 1;
			.@current_min = 1;
		}

		dispbottom "[KoE]: Participation Time: " + .@current_min + " / " + .timepart + " minute(s).";

		if (.@current_min >= .timepart)
			dispbottom "[KoE]: Your participation reward requirement is complete.";
		else
			dispbottom "[KoE]: You need " + (.timepart - .@current_min) + " more minute(s) for participation reward.";
	} else
		set @lastmap$, "";
	end;

OnInit:
	// Options:
	// 1 = enable guild reward system
	// Removed option 4 so reward is given to the whole winning guild, not only 1 player.
	set .Options, 1;

	// Participation reward requires 10 active minutes.
	set .timepart, 10;
	set .Participation, 1;

	// Anti-AFK idle limit: 4 minutes.
	set .idle_limit, 240;

	// Reward format:
	// KoE Points + Ygg Berry Box (33021) + Ygg Seed Box (33022)
	// Latest reward setting:
	// Participation = 10 + 10 + 10
	// Defender      = 10 + 10 + 10
	// Holder        = 20 + 20 + 20
	set .part_points, 10;
	set .part_berry, 10;
	set .part_seed, 10;

	set .defender_points, 10;
	set .defender_berry, 10;
	set .defender_seed, 10;

	set .holder_points, 20;
	set .holder_berry, 20;
	set .holder_seed, 20;


	setarray .maps$[0],
		"koe_rachel","KoE Rachel",
		"koe_hyper","KoE Hyper",
		"koe_cmd","KoE Comodo",
		"koe_prt","KoE Prontera",
		"koe_gef","KoE Geffen",
		"koe_moc","KoE Morocc";

	set .Emperium_id, 28003;
	set .Barricade_id, 28004;

	setarray .koe_rachel_emp[0],50,50;
	setarray .koe_hyper_emp[0],99,99;
	setarray .koe_cmd_emp[0],141,162;
	setarray .koe_prt_emp[0],171,260;
	setarray .koe_gef_emp[0],91,127;
	setarray .koe_moc_emp[0],91,127;

	setarray .koe_rachel_guardian[0],28000,42,57,28001,58,41;
	setarray .koe_hyper_guardian[0],28000,100,90,28001,100,109;
	setarray .koe_cmd_guardian[0],28000,122,149,28001,160,150;
	setarray .koe_prt_guardian[0],28000,164,231,28001,178,231;
	setarray .koe_gef_guardian[0],28000,110,99,28001,71,94;
	setarray .koe_moc_guardian[0],28000,110,99,28001,71,94;

	set .enableBarricade, 1;

	setarray .warpbreak_koe_rachel[0],42,50;
	setarray .walkable_koe_rachel[0],46,53,53,46;
	setarray .barx_koe_rachel[0],45,45,48,48,51,51,54,54;
	setarray .bary_koe_rachel[0],51,48,54,45,54,45,51,48;

	setarray .warpbreak_koe_hyper[0],112,99;
	setarray .walkable_koe_hyper[0],94,105,105,94;
	setarray .barx_koe_hyper[0],97,102,105,105,102,97,94,94;
	setarray .bary_koe_hyper[0],94,94,97,102,105,105,102,97;

	setarray .warpbreak_koe_cmd[0],140,180;
	setarray .walkable_koe_cmd[0],139,164,143,161;
	setarray .barx_koe_cmd[0],138,137,138,140,142,144,145,144;
	setarray .bary_koe_cmd[0],164,162,160,159,159,160,162,164;

	setarray .warpbreak_koe_prt[0],171,232;
	setarray .walkable_koe_prt[0],168,263,175,256;
	setarray .barx_koe_prt[0],169,174,174,169,167,167,176,176;
	setarray .bary_koe_prt[0],255,255,264,264,261,258,258,261;

	setarray .warpbreak_koe_gef[0],92,113;
	setarray .walkable_koe_gef[0],88,131,95,124;
	setarray .barx_koe_gef[0],89,94,94,89;
	setarray .bary_koe_gef[0],123,123,132,132;

	setarray .warpbreak_koe_moc[0],92,113;
	setarray .walkable_koe_moc[0],88,131,95,124;
	setarray .barx_koe_moc[0],89,94,94,89;
	setarray .bary_koe_moc[0],123,123,132,132;

	for (.@i = 0; .@i < getarraysize(.maps$); .@i += 2) {
		setmapflag .maps$[.@i], mf_gvg;
		setmapflag .maps$[.@i], mf_nobranch;
		setmapflag .maps$[.@i], mf_nomemo;
		setmapflag .maps$[.@i], mf_nopenalty;
		setmapflag .maps$[.@i], mf_nosave;
		setmapflag .maps$[.@i], mf_noteleport;
		setmapflag .maps$[.@i], mf_nowarp;
		setmapflag .maps$[.@i], mf_nowarpto;
		setmapflag .maps$[.@i], mf_loadevent;
		setmapflag .maps$[.@i], mf_nomobloot;
		setmapflag .maps$[.@i], mf_kingofemp;
	}

	for (.@i = 0; .@i < getarraysize(.maps$); .@i += 2)
		killmonsterall .maps$[.@i];
	end;
}
