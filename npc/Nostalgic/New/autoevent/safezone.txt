-	script	MG_SafeZone	FAKE_NPC,{

OnInit:
	.evtName$ = "Safe Zone";
	setwall("safez_ev", 48, 66, 8, DIR_EAST, false, "#safezWallA");
	.group_lv = 100;
	end;

OnClearDupe:
	duplicate_remove("SafeZone");
	end;

OnStart:
	$EVT_SAFE = 2;
	.ROUND = 0;
	setmapflag("safez_ev", MF_NOSKILL);
	mapannounce("safez_ev", "["+.evtName$+"] This is how the event works.", bc_npc|bc_blue);
	sleep 2000;
	mapannounce("safez_ev", "["+.evtName$+"] Make sure that your effects are active (/effect ON).", bc_npc|bc_blue);
	sleep 2000;
	mapannounce("safez_ev", "["+.evtName$+"] Enter a 'Safety Wall' to surive.", bc_npc|bc_blue);
	sleep 2000;
	mapannounce("safez_ev", "["+.evtName$+"] This will run until only one player survives.", bc_npc|bc_blue);
	sleep 2000;
	setnpctimer(300000);
	startnpctimer();
	end;

OnTimer310000:
	donpcevent(strnpcinfo(0)+"::OnSafetyWall");
	end;

OnSafetyWall:
	while ($EVT_SAFE == 2) {
		stopnpctimer();
		.ROUND++;
		if (checkwall("#safezWallA")) delwall("#safezWallA");
		deletearray(.@units);
		.@cnt = getmapunits(BL_PC, "safez_ev", .@units[0]);
		for (.@i=0;.@i<.@cnt; .@i++) {
			if (attachrid(.@units[.@i]))
				warp("safez_ev", rand(50, 53), rand(71, 73));
		}
		detachrid();

		deletearray(.@units);
		.@cnt = getmapunits(BL_PC, "safez_ev", .@units[0]);
		.@gm_cnt = 0;

		freeloop(true);
		for (.@i=0; .@i<.@cnt; .@i++) {
			if (attachrid(.@units[.@i])) {
				if (getgroupid() >= .group_lv)
					++.@gm_cnt;
			}
		}
		detachrid();
		.@cnt2 = getmapusers("safez_ev") - .@gm_cnt;
		.@sw_cnt = .@cnt2 - (.@cnt2 * 10 / 100);
		if (.@cnt2 <= 8)
			.@sw_cnt = .@cnt2/2;
		if (.@sw_cnt < 1)
			.@sw_cnt = 1;
		if (.@sw_cnt > 50)
			.@sw_cnt = 50;

		for (.@i=0; .@i<.@sw_cnt; .@i++) {
			while (true) {
				.@x = rand(40,63);
				.@y = rand(32,55);
				if (checkcell("safez_ev",.@x,.@y,cell_chknpc))
					continue;
				break;
			}
			.@ex_name$[.@i] = duplicate("SafeZone", "safez_ev", .@x, .@y, sprintf("SafeZone%d", .@i), HIDDEN_WARP_NPC, 0, 1, 1);
		}

		sleep(300);
		for (.@i=0; .@i<.@sw_cnt; .@i++)
			donpcevent(.@ex_name$[.@i]+"::OnShowSafetyWall");

		removemapflag("safez_ev", MF_NOSKILL);
		sleep(10000);
		setmapflag("safez_ev", MF_NOSKILL);
		donpcevent(strnpcinfo(0)+"::OnClearDupe");

		deletearray(.@units);
		.@cnt = getmapunits(BL_PC, "safez_ev", .@units[0]);
		for (.@i=0; .@i<.@cnt; .@i++) {
			if (attachrid(.@units[.@i])) {
				if (getgroupid() >= .group_lv)
					continue;
				if (@safety_safe + 13 > gettimetick(2))
					continue;
				specialeffect2(EF_ARROWSTORM_STR, AREA, rid2name(.@units[.@i]));
				dispbottom(sprintf("[%s] You are out! Better luck next time.", .evtName$));
				if ($EVT_SAFE_CONSO_POINT) {
					F_PointGain(1, $EVT_SAFE_CONSO_POINT, .evtName$);
					F_ShowCurrency("#EVENTPOINT", "Event", $EVT_SAFE_CONSO_POINT);
					EVENTWINS += $EVT_SAFE_CONSO_POINT;
					EVENTWINS_ZOMBIE += $EVT_SAFE_CONSO_POINT;
				}
				unitkill(.@units[.@i]);
			}
		}
		detachrid();

		sleep(1300);
		.@cnt = getmapunits(BL_PC, "safez_ev", .@units[0]);
		for (.@i=0; .@i<.@cnt; .@i++) {
			if (attachrid(.@units[.@i])) {
				if (getgroupid() >= .group_lv)
					continue;
				if (@safety_safe + 13 > gettimetick(2))
					continue;
				warp("prontera", 149, 156);
			}
		}

		sleep(500);
		deletearray(.@units);
		.@cnt = getmapunits(BL_PC, "safez_ev", .@units[0]);
		.@cnt2 = 0;
		for (.@i=0; .@i<.@cnt; .@i++) {
			if (attachrid(.@units[.@i])) {
				if (getgroupid() >= .group_lv)
					continue;
				++.@cnt2;
			}
		}
		detachrid();

		if (.@cnt2 < 2) {
			for (.@i=0; .@i<.@cnt; .@i++) {
				if (attachrid(.@units[.@i])) {
					if (getgroupid() >= .group_lv)
						continue;
					announce("["+.evtName$+"] "+strcharinfo(0)+" has won the event!",bc_all|bc_blue);
					$EVT_SAFE_WINNER$ = strcharinfo(0);
					if (getarraysize($EVT_SAFE_ITEM) > 1) {
						for (.@i=0; .@i<getarraysize($EVT_SAFE_ITEM); .@i+=2)
							getitem($EVT_SAFE_ITEM[.@i], $EVT_SAFE_ITEM[.@i+1]);
					}
					if ($EVT_SAFE_POINT > 0) {
						F_PointGain(0, $EVT_SAFE_POINT, .evtName$);
						F_ShowCurrency("#EVENTPOINT", "Event", $EVT_SAFE_POINT);
						EVENTWINS += $EVT_SAFE_POINT;
						EVENTWINS_BOMB += $EVT_SAFE_POINT;
					}
				}
				break;
			}
			detachrid();
			donpcevent(strnpcinfo(0)+"::OnStop");
			end;
		}
		freeloop(false);

		mapannounce("safez_ev", "["+.evtName$+"] Next Round in 5 seconds!", bc_npc|bc_blue);
		setnpctimer(305000);
		startnpctimer();
		setwall("safez_ev", 48, 66, 8, DIR_EAST, false, "#safezWallA");
		end;
	}
	donpcevent(strnpcinfo(0)+"::OnStop");
	end;

OnStop:
	stopnpctimer();
	setwall("safez_ev", 48, 66, 8, DIR_EAST, false, "#safezWallA");
	$EVT_SAFE = 0;
	.ROUND = 0;
	mapwarp("safez_ev", "prontera", 149, 156);
	end;
}

safez_ev,0,0,0	script	SafeZone::SafeZone	HIDDEN_WARP_NPC,1,1,{
	end;

OnTouch:
	if (@safety_safe + 13 > gettimetick(2)) end;
	@safety_safe = gettimetick(2);
	unitstopwalk(getcharid(3), USW_FORCE_STOP);
	dispbottom("[Safe Zone] You are safe!");
	unitskilluseid(getcharid(3), "TF_HIDING", 1);
	specialeffect2(EF_PORTAL4, AREA);
	duplicate_remove(strnpcinfo(3));
	end;

OnShowSafetyWall:
	getmapxy(.@map$, .@x, .@y, BL_NPC);
	unitskillusepos(getnpcid(0), "MG_SAFETYWALL", 10, .@x, .@y);
	end;
}
