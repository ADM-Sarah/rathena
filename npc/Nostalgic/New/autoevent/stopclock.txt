stc_ev,48,47,4	script	Stop The Clock	ALARM,{

	if (!$EVT_STOPCLOCK_CLICK) end;
	if (#STOPCLOCK) {
		message(strcharinfo(0), "You already stopped the clock.");
		end;
	}
	#STOPCLOCK = 1;
	@STOPCLOCK_TICK = .i;
	.stopclick_r[getarraysize(.stopclick_r)] = getcharid(3);
	message(strcharinfo(0), sprintf("You stopped the clock at %d.", @STOPCLOCK_TICK));
	if (.stopLowest > @STOPCLOCK_TICK) {
		.stopLowest = @STOPCLOCK_TICK;
		.winner$ = strcharinfo(0);
		.winnerAID = getcharid(3);
	}
	end;

OnStart:
	if (!$EVT_STOPCLOCK_CLICK) end;
	.stopLowest = 1000;
	.winner$ = "";
	.winnerAID = 0;
	for (.i = 1000; .i > 300; .i -= 100) {
		announce(sprintf(":: %d ::", .i), bc_blue|bc_area|bc_npc);
		specialeffect(EF_BASH3D2);
		sleep(1000);
	}
	for (.i = 300; .i > 50; .i -= 10) {
		announce(sprintf(":: %d ::", .i), bc_blue|bc_area|bc_npc);
		specialeffect(EF_BASH3D2);
		sleep(100);
	}
	for (.i = 50; .i > 0; .i--) {
		announce(sprintf(":: %d ::", .i), bc_blue|bc_area|bc_npc);
		specialeffect(EF_BASH3D2);
		sleep(10);
	}
	for (.@i=0; .@i<getarraysize(.stopclick_r); ++.@i) {
		if (attachrid(.stopclick_r[.@i]))
			#STOPCLOCK = 0;
		detachrid();
	}
	donpcevent("MG_StopTheClock::OnStop");
	end;

OnReset:
	.stopLowest = 1000;
	.winner$ = "";
	.winnerAID = 0;
	end;
}

-	script	MG_StopTheClock	FAKE_NPC,{
	end;

OnInit:
	.GRP_ID = 70;
	.evtName$ = "Stop the Clock";
	disablenpc("Stop The Clock");
	if ($EVT_STOPCLOCK)
		donpcevent(strnpcinfo(0)+"::OnStart");
	end;

OnStart:
	setmapflag("stc_ev", MF_NOCOMMAND, .GRP_ID);
	enablenpc("Stop The Clock");
	initnpctimer();
	end;

OnTimer10000:
	mapannounce("stc_ev", "["+.evtName$+"] Prepare to click me with precision.", bc_all|bc_blue);
	end;

OnTimer15000:
	mapannounce("stc_ev", "["+.evtName$+"] Make sure to hit me on the right moment.", bc_all|bc_blue);
	end;

OnTimer20000:
	mapannounce("stc_ev", "["+.evtName$+"] In 5...", bc_all|bc_blue);
	end;

OnTimer21000:
	mapannounce("stc_ev", "["+.evtName$+"] 4...", bc_all|bc_blue);
	end;

OnTimer22000:
	mapannounce("stc_ev", "["+.evtName$+"] 3...", bc_all|bc_blue);
	end;

OnTimer23000:
	mapannounce("stc_ev", "["+.evtName$+"] 2...", bc_all|bc_blue);
	end;

OnTimer24000:
	mapannounce("stc_ev", "["+.evtName$+"] 1...", bc_all|bc_blue);
	end;

OnTimer25000:
	stopnpctimer();
	mapannounce("stc_ev", "["+.evtName$+"] Start!", bc_all|bc_blue);
	$EVT_STOPCLOCK = 1;
	$EVT_STOPCLOCK_CLICK = 1;
	donpcevent("Stop The Clock::OnStart");
	end;

OnStop:
	stopnpctimer();
	$EVT_STOPCLOCK = 0;
	$EVT_STOPCLOCK_CLICK = 0;
	disablenpc("Stop The Clock");
	.@name$ = getvariableofnpc(.winner$, "Stop The Clock");
	.@aid = getvariableofnpc(.winnerAID, "Stop The Clock");
	.@tick = getvariableofnpc(.stopLowest, "Stop The Clock");
	if (.@name$ == "" && $EVT_STOPCLOCK_CLICK == 1) {
		mapannounce("stc_ev", "["+.evtName$+"] Nobody hits the clock at the right moment. No winners on this mini-game.", bc_all|bc_blue);
		removemapflag("stc_ev", MF_NOCOMMAND);
		sleep(1000);
		mapwarp("stc_ev", "prontera", 149, 156);
		donpcevent("Stop The Clock::OnReset");
		end;
	}
	if (attachrid(.@aid)) {
		mapannounce("stc_ev", "["+.evtName$+"] "+strcharinfo(0)+" won the mini-game at stopped at "+.@tick+".", bc_all|bc_blue);
		$EVT_STOPCLOCK_WINNER$ = strcharinfo(0);
		if (getarraysize($EVT_STOPCLOCK_ITEM) > 1) {
			for (.@i=0; .@i<getarraysize($EVT_STOPCLOCK_ITEM); .@i+=2)
				getitem($EVT_STOPCLOCK_ITEM[.@i], $EVT_STOPCLOCK_ITEM[.@i+1]);
		}
		if ($EVT_STOPCLOCK_POINT > 0) {
			F_PointGain(0, $EVT_STOPCLOCK_POINT, .evtName$);
			F_ShowCurrency("#EVENTPOINT", "Event", $EVT_STOPCLOCK_POINT);
			EVENTWINS += $EVT_STOPCLOCK_POINT;
			EVENTWINS_CLOCK += $EVT_STOPCLOCK_POINT;
		}
	}
	.@cnt = getmapunits(BL_PC, "stc_ev", .@units[0]);
	if (.@cnt && $EVT_STOPCLOCK_CONSO_POINT) {
		for (.@i=0; .@i<.@cnt; .@i++) {
			if (.@aid == .@units[.@i])
				continue;
			if (attachrid(.@units[.@i])) {
				F_PointGain(1, $EVT_STOPCLOCK_CONSO_POINT, .evtName$);
				F_ShowCurrency("#EVENTPOINT", "Event", $EVT_STOPCLOCK_CONSO_POINT);
				EVENTWINS += $EVT_STOPCLOCK_CONSO_POINT;
				EVENTWINS_CLOCK += $EVT_STOPCLOCK_CONSO_POINT;
			}
			detachrid();
		}
	}
	removemapflag("stc_ev", MF_NOCOMMAND);
	sleep(1000);
	mapwarp("stc_ev", "prontera", 149, 156);
	donpcevent("Stop The Clock::OnReset");
	end;
}
