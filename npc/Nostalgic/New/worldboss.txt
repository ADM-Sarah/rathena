//===== World Boss MVP Counter Function =======================
//= Used by normal MVP kills, Bloody Branch, and custom MVP summoner.
//============================================================
function	script	F_WorldBoss_AddMvpKill	{
	.@amount = getarg(0,1);
	if (.@amount < 1) .@amount = 1;

	if (getvariableofnpc(.status,"WORLDBOSS") || getvariableofnpc(.WorldBossReady,"WORLDBOSS"))
		return;

	set $KilledMvp, $KilledMvp + .@amount;

	if (getvariableofnpc(.MvpKills,"WORLDBOSS") > 0 && $KilledMvp >= getvariableofnpc(.MvpKills,"WORLDBOSS")) {
		set getvariableofnpc(.WorldBossReady,"WORLDBOSS"), 1;
		donpcevent "WORLDBOSS::OnCheckKills";
	}
	return;
}


//============================================================
// World Boss Entry Checker - IP + Gepard Unique ID
// FULL FIX: LIVE REGISTRY CHECK, NO MORE `char.last_map` DEPENDENCY
// - Blocks entry if same IP is already inside worldboss map.
// - Blocks entry if same Gepard unique_id/device is already inside.
// - Uses live registry + attachrid verification, because SQL last_map can delay/leak.
// - This is a LIVE map check
//============================================================
function	script	F_WB_CleanupRegistry	{
	.@map$ = getarg(0, "worldboss");
	.@orig_aid = getcharid(3);

	for (.@i = getarraysize($@WB_AID) - 1; .@i >= 0; .@i--) {
		.@remove = 0;

		if ($@WB_AID[.@i] <= 0) {
			.@remove = 1;
		}
		else if (!attachrid($@WB_AID[.@i])) {
			.@remove = 1;
		}
		else if (strcharinfo(3) != .@map$) {
			.@remove = 1;
		}
		else {
			// Refresh data while player is confirmed alive inside map.
			$@WB_CID[.@i] = getcharid(0);
			$@WB_NAME$[.@i] = strcharinfo(0);
			$@WB_IP$[.@i] = getcharip();
			$@WB_UID$[.@i] = get_unique_id();
		}

		if (.@remove) {
			deletearray $@WB_AID[.@i], 1;
			deletearray $@WB_CID[.@i], 1;
			deletearray $@WB_NAME$[.@i], 1;
			deletearray $@WB_IP$[.@i], 1;
			deletearray $@WB_UID$[.@i], 1;
		}
	}

	if (.@orig_aid > 0)
		attachrid(.@orig_aid);
	return;
}

function	script	F_WB_CheckEntry	{
	.@map$ = getarg(0, "worldboss");
	.@orig_aid = getcharid(3);
	.@orig_cid = getcharid(0);
	.@my_ip$ = getcharip();
	.@my_uid$ = get_unique_id();

	$@WB_BlockReason$ = "";

	// Clean old/logout/warped entries first.
	callfunc "F_WB_CleanupRegistry", .@map$;

	for (.@i = 0; .@i < getarraysize($@WB_AID); .@i++) {
		// Ignore current character if already registered.
		if ($@WB_AID[.@i] == .@orig_aid && $@WB_CID[.@i] == .@orig_cid)
			continue;

		if ($@WB_IP$[.@i] != "" && $@WB_IP$[.@i] == .@my_ip$) {
			$@WB_BlockReason$ = "[World Boss] Only 1 player per IP can enter the World Boss Area. Duplicate IP detected with player: "+$@WB_NAME$[.@i]+".";
			return 0;
		}

		if (.@my_uid$ != "" && $@WB_UID$[.@i] != "" && $@WB_UID$[.@i] == .@my_uid$) {
			$@WB_BlockReason$ = "[World Boss] Only 1 player per device can enter the World Boss Area. Duplicate device detected with player: "+$@WB_NAME$[.@i]+".";
			return 0;
		}
	}

	return 1;
}

function	script	F_WB_RegisterEntry	{
	.@map$ = getarg(0, "worldboss");
	.@aid = getcharid(3);
	.@cid = getcharid(0);

	callfunc "F_WB_CleanupRegistry", .@map$;

	// Update existing record if same character already registered.
	for (.@i = 0; .@i < getarraysize($@WB_AID); .@i++) {
		if ($@WB_AID[.@i] == .@aid && $@WB_CID[.@i] == .@cid) {
			$@WB_NAME$[.@i] = strcharinfo(0);
			$@WB_IP$[.@i] = getcharip();
			$@WB_UID$[.@i] = get_unique_id();
			return;
		}
	}

	.@n = getarraysize($@WB_AID);
	$@WB_AID[.@n] = .@aid;
	$@WB_CID[.@n] = .@cid;
	$@WB_NAME$[.@n] = strcharinfo(0);
	$@WB_IP$[.@n] = getcharip();
	$@WB_UID$[.@n] = get_unique_id();
	return;
}

function	script	F_WB_UnregisterEntry	{
	.@aid = getcharid(3);
	.@cid = getcharid(0);

	for (.@i = getarraysize($@WB_AID) - 1; .@i >= 0; .@i--) {
		if ($@WB_AID[.@i] == .@aid && $@WB_CID[.@i] == .@cid) {
			deletearray $@WB_AID[.@i], 1;
			deletearray $@WB_CID[.@i], 1;
			deletearray $@WB_NAME$[.@i], 1;
			deletearray $@WB_IP$[.@i], 1;
			deletearray $@WB_UID$[.@i], 1;
			return;
		}
	}
	return;
}

function	script	F_WB_ClearRegistry	{
	deletearray $@WB_AID[0], getarraysize($@WB_AID);
	deletearray $@WB_CID[0], getarraysize($@WB_CID);
	deletearray $@WB_NAME$[0], getarraysize($@WB_NAME$);
	deletearray $@WB_IP$[0], getarraysize($@WB_IP$);
	deletearray $@WB_UID$[0], getarraysize($@WB_UID$);
	return;
}

// Auto Kick AFK / Early Entry
// - If player enters worldboss map while event is NOT running
//   and countdown is NOT active, they will be warped out after 5 sec.
// - During 5-minute countdown (.WorldBossReady = 1) players can stay.
// - During active event (.status = 1) players can stay.

OnPCLoadMapEvent:
	if(strcharinfo(3) != .Map$[0])
		end;

	// IP + Device live check.
	// Event start or not start, if same IP/device already inside the map, this player will be warped out.
	if (!callfunc("F_WB_CheckEntry", .Map$[0])) {
		dispbottom $@WB_BlockReason$;
		warp "SavePoint",0,0;
		end;
	}

	// Register this player as ACTIVE inside World Boss map.
	// This fixes multiple ID leak from SQL last_map delay.
	callfunc "F_WB_RegisterEntry", .Map$[0];

	if(.status || .WorldBossReady)
		end;

	dispbottom "[World Boss] The event is not active yet. You will be warped out in 5 seconds.";
	sleep2 5000;

	if(strcharinfo(3) == .Map$[0] && !.status && !.WorldBossReady){
		dispbottom "[World Boss] Please come back when the Boss Raid countdown/event is active.";
		callfunc "F_WB_UnregisterEntry";
		warp "SavePoint",0,0;
	}
	end;

OnPCLogoutEvent:
	if(strcharinfo(3) == .Map$[0])
		callfunc "F_WB_UnregisterEntry";
	end;

OnNPCKillEvent:
	if(.status || .WorldBossReady) end;

	.@is_mvp = 0;

	if(getmonsterinfo(killedrid, MOB_MVPEXP) > 0)
		.@is_mvp = 1;

	for(.@i = 0; .@i < getarraysize(.ForceCountMVP); .@i++){
		if(killedrid == .ForceCountMVP[.@i]){
			.@is_mvp = 1;
			break;
		}
	}

	if(!.@is_mvp) end;

	callfunc "F_WorldBoss_AddMvpKill", 1;
end;
